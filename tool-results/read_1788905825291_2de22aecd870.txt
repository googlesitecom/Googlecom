   188→   188→// Daño efectivo (usado por el servidor)
   189→   189→export function computeDamage(w: WeaponConfig, part: BodyPart, distance: number, victimArmor: number): number {
   190→   190→  let dmg = w.damage
   191→   191→  if (part === 'head') dmg *= w.headMult
   192→   192→  else if (part === 'legs') dmg *= w.legMult
   193→   193→  // caída por distancia
   194→   194→  if (distance > w.falloffStart) {
   195→   195→    const t = Math.min(1, (distance - w.falloffStart) / Math.max(1, w.falloffEnd - w.falloffStart))
   196→   196→    dmg *= 1 - (1 - w.falloffMin) * t
   197→   197→  }
   198→   198→  // armadura (la cabeza sin casco ya multiplicó; simplificamos: armadura protege cuerpo y cabeza)
   199→   199→  if (victimArmor > 0 && part !== 'legs') {
   200→   200→    dmg *= 1 - w.armorAbsorb
   201→   201→  }
   202→   202→  return Math.round(dmg)
   203→   203→}

   272→   272→function hut(cx: number, cz: number, doorSide: 'n' | 's' | 'e' | 'w', door2?: 'n' | 's' | 'e' | 'w') {
   273→   273→  const S = 4, H = 3, T2 = 0.4, G = 1.5
   274→   274→  const sides: ('n' | 's' | 'e' | 'w')[] = ['n', 's', 'e', 'w']
   275→   275→  for (const s of sides) {
   276→   276→    const hasDoor = s === doorSide || s === door2
   277→   277→    const segLen = hasDoor ? (S - G) : (2 * S)
   278→   278→    if (s === 'n') {
   279→   279→      if (hasDoor) {
   280→   280→        B(cx - (G + segLen / 2), H / 2, cz - S, segLen, H, T2, 'sand')
   281→   281→        B(cx + (G + segLen / 2), H / 2, cz - S, segLen, H, T2, 'sand')
   282→   282→        B(cx, H - 0.4, cz - S, G * 2, 0.8, T2, 'sand')
   283→   283→      } else B(cx, H / 2, cz - S, 2 * S, H, T2, 'sand')
   284→   284→    }
   285→   285→    if (s === 's') {
   286→   286→      if (hasDoor) {
   287→   287→        B(cx - (G + segLen / 2), H / 2, cz + S, segLen, H, T2, 'sand')
   288→   288→        B(cx + (G + segLen / 2), H / 2, cz + S, segLen, H, T2, 'sand')
   289→   289→        B(cx, H - 0.4, cz + S, G * 2, 0.8, T2, 'sand')
   290→   290→      } else B(cx, H / 2, cz + S, 2 * S, H, T2, 'sand')
   291→   291→    }
   292→   292→    if (s === 'e') {
   293→   293→      if (hasDoor) {
   294→   294→        B(cx + S, H / 2, cz - (G + segLen / 2), T2, H, segLen, 'sand')
   295→   295→        B(cx + S, H / 2, cz + (G + segLen / 2), T2, H, segLen, 'sand')
   296→   296→        B(cx + S, H - 0.4, cz, T2, 0.8, G * 2, 'sand')
   297→   297→      } else B(cx + S, H / 2, cz, T2, H, 2 * S, 'sand')
   298→   298→    }
   299→   299→    if (s === 'w') {
   300→   300→      if (hasDoor) {
   301→   301→        B(cx - S, H / 2, cz - (G + segLen / 2), T2, H, segLen, 'sand')
   302→   302→        B(cx - S, H / 2, cz + (G + segLen / 2), T2, H, segLen, 'sand')
   303→   303→        B(cx - S, H - 0.4, cz, T2, 0.8, G * 2, 'sand')
   304→   304→      } else B(cx - S, H / 2, cz, T2, H, 2 * S, 'sand')
   305→   305→    }
   306→   306→  }

   491→   491→/** ¿Choca el segmento p→q con alguna caja? (para línea de visión) */
   492→   492→export function segmentBlocked(px: number, py: number, pz: number, qx: number, qy: number, qz: number, boxes: AABB[] = MAP_AABBS): boolean {
   493→   493→  const dx = qx - px, dy = qy - py, dz = qz - pz
   494→   494→  for (let i = 0; i < boxes.length; i++) {
   495→   495→    const b = boxes[i]
   496→   496→    let tmin = 0, tmax = 1
   497→   497→    // X
   498→   498→    if (Math.abs(dx) < 1e-9) {
   499→   499→      if (px < b.minX || px > b.maxX) continue
   500→   500→    } else {
   501→   501→      let t1 = (b.minX - px) / dx, t2 = (b.maxX - px) / dx
   502→   502→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   503→   503→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   504→   504→      if (tmin > tmax) continue
   505→   505→    }
   506→   506→    // Y
   507→   507→    if (Math.abs(dy) < 1e-9) {
   508→   508→      if (py < b.minY || py > b.maxY) continue
   509→   509→    } else {
   510→   510→      let t1 = (b.minY - py) / dy, t2 = (b.maxY - py) / dy
   511→   511→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   512→   512→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   513→   513→      if (tmin > tmax) continue
   514→   514→    }
   515→   515→    // Z
   516→   516→    if (Math.abs(dz) < 1e-9) {
   517→   517→      if (pz < b.minZ || pz > b.maxZ) continue
   518→   518→    } else {
   519→   519→      let t1 = (b.minZ - pz) / dz, t2 = (b.maxZ - pz) / dz
   520→   520→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   521→   521→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   522→   522→      if (tmin > tmax) continue
   523→   523→    }
   524→   524→    return true
   525→   525→  }
   526→   526→  return false
   527→   527→}
