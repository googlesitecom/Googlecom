   236→/** Convierte un KeyboardEvent.code a etiqueta legible (KeyW → W, Digit1 → 1…) */
   237→export function keyLabel(code: string): string {
   238→  if (!code) return '—'
   239→  if (code.startsWith('Key')) return code.slice(3)
   240→  if (code.startsWith('Digit')) return code.slice(5)
   241→  if (code.startsWith('Numpad')) return 'NUM ' + code.slice(6)
   242→  if (code.startsWith('Arrow')) {
   243→    const names: Record<string, string> = { Up: '↑', Down: '↓', Left: '←', Right: '→' }
   244→    return names[code.slice(5)] ?? code
   245→  }
   246→  const names: Record<string, string> = {
   247→    Space: 'ESPACIO', ShiftLeft: 'MAYÚS IZQ', ShiftRight: 'MAYÚS DER',
   248→    ControlLeft: 'CTRL IZQ', ControlRight: 'CTRL DER',
   249→    AltLeft: 'ALT', AltRight: 'ALT GR', Enter: 'ENTER', Tab: 'TAB',
   250→    CapsLock: 'BLOQ MAYÚS', Backquote: '`', Minus: '-', Equal: '=',
   251→    BracketLeft: '[', BracketRight: ']', Semicolon: ';', Quote: "'",
   252→    Comma: ',', Period: '.', Slash: '/', Backslash: '\\',
   253→  }
   254→  return names[code] ?? code.toUpperCase()
   255→}

   266→/** Daño efectivo de un arma (sin escudo: la absorción la aplica la simulación) */
   267→export function computeDamage(w: WeaponConfig, part: BodyPart, distance: number): number {
   268→  let dmg = w.damage
   269→  if (part === 'head') dmg *= w.headMult
   270→  else if (part === 'legs') dmg *= w.legMult
   271→  if (distance > w.falloffStart) {
   272→    const t = Math.min(1, (distance - w.falloffStart) / Math.max(1, w.falloffEnd - w.falloffStart))
   273→    dmg *= 1 - (1 - w.falloffMin) * t
   274→  }
   275→  return Math.round(dmg)
   276→}

   345→// --- Muros con puerta central (edificios genéricos) ---
   346→/** Muro a lo largo del eje `ax`, en la coordenada fija `at`, desde `from` hasta `to`, con puerta centrada de ancho 2·gateHalf */
   347→function gatedWall(ax: 'x' | 'z', at: number, from: number, to: number, H: number, T: number, mat: MatKey, gateHalf = 1.6, doorH = 0) {
   348→  const lo = Math.min(from, to), hi = Math.max(from, to)
   349→  const mid = (lo + hi) / 2
   350→  const mk = (c: number, len: number) => {
   351→    if (len <= 0.05) return
   352→    if (ax === 'x') B(c, H / 2, at, len, H, T, mat)
   353→    else B(at, H / 2, c, T, H, len, mat)
   354→  }
   355→  mk((lo + (mid - gateHalf)) / 2, (mid - gateHalf) - lo)
   356→  mk((hi + (mid + gateHalf)) / 2, hi - (mid + gateHalf))
   357→  // dintel sobre la puerta (doorH = altura libre de la puerta)
   358→  if (gateHalf > 0 && H > doorH + 0.6) {
   359→    const lintelH = H - Math.max(doorH, 0.01)
   360→    if (lintelH > 0.15) {
   361→      if (ax === 'x') B(mid, H - lintelH / 2, at, gateHalf * 2, lintelH, T, mat)
   362→      else B(at, H - lintelH / 2, mid, T, lintelH, gateHalf * 2, mat)
   363→    }
   364→  }
   365→}

   367→// ------------------------------------------------------------
   368→// Edificios con interior: muros con puerta + banda de ventanas
   369→// ------------------------------------------------------------
   370→/** Rotaciones: la fachada (puerta) del edificio local mira a -z; `f` gira el edificio */
   371→const ROT = { N: 0, E: Math.PI / 2, S: Math.PI, W: Math.PI * 1.5 } as const
   372→type Facing = keyof typeof ROT
   373→
   374→/** Caja en coordenadas locales del edificio, girada según `f` */
   375→function BR(cx: number, cz: number, f: Facing, lx: number, ly: number, lz: number, w: number, h: number, d: number, mat: MatKey): void {
   376→  const ang = ROT[f]
   377→  const c = Math.cos(ang), s = Math.sin(ang)
   378→  const rx = lx * c - lz * s
   379→  const rz = lx * s + lz * c
   380→  const swap = Math.abs(s) > 0.5
   381→  B(cx + rx, ly, cz + rz, swap ? d : w, h, swap ? w : d, mat)

   384→interface WallOpts {
   385→  H?: number; T?: number; y0?: number
   386→  door?: number; doorHalf?: number; doorH?: number
   387→  wins?: number[]; winW?: number; bandLo?: number; bandHi?: number
   388→}

   390→/**
   391→ * Muro de edificio en coords. locales. `axis 'z'`: corre a lo largo de x en z=at;
   392→ * `axis 'x'`: corre a lo largo de z en x=at. Puerta = hueco a toda altura con dintel;
   393→ * ventanas = banda practicable (se puede disparar a través) entre zócalo y franja superior.
   394→ */
   395→function wallL(cx: number, cz: number, f: Facing, axis: 'x' | 'z', at: number, from: number, to: number, mat: MatKey, opts: WallOpts): void {
   396→  const H = opts.H ?? 3.3
   397→  const T = opts.T ?? 0.35
   398→  const y0 = opts.y0 ?? 0
   399→  const bandLo = y0 + (opts.bandLo ?? 1.45)
   400→  const bandHi = y0 + (opts.bandHi ?? 2.25)
   401→  const winW = opts.winW ?? 1.3
   402→  const lo = Math.min(from, to), hi = Math.max(from, to)
   403→  const put = (c: number, len: number, ya: number, yb: number) => {
   404→    if (len <= 0.06 || yb - ya <= 0.06) return
   405→    const cy = (ya + yb) / 2
   406→    if (axis === 'z') BR(cx, cz, f, c, cy, at, len, yb - ya, T, mat)
   407→    else BR(cx, cz, f, at, cy, c, T, yb - ya, len, mat)
   408→  }
   409→  let solids: [number, number][] = [[lo, hi]]
   410→  if (opts.door !== undefined) {
   411→    const dh = opts.doorHalf ?? 1.05
   412→    const doorH = y0 + (opts.doorH ?? 2.15)
   413→    const dLo = opts.door - dh, dHi = opts.door + dh
   414→    solids = []
   415→    if (dLo - lo > 0.05) solids.push([lo, dLo])
   416→    if (hi - dHi > 0.05) solids.push([dHi, hi])
   417→    put(opts.door, dh * 2, doorH, y0 + H) // dintel sobre la puerta
   418→  }
   419→  for (const [a, b] of solids) {
   420→    put((a + b) / 2, b - a, y0, bandLo)               // zócalo
   421→    put((a + b) / 2, b - a, bandHi, y0 + H)           // franza superior
   422→    const gaps: [number, number][] = []               // huecos de ventana
   423→    for (const w of opts.wins ?? []) {
   424→      const g0 = w - winW / 2, g1 = w + winW / 2
   425→      if (g0 > a + 0.3 && g1 < b - 0.3) gaps.push([g0, g1])
   426→    }
   427→    gaps.sort((p, q) => p[0] - q[0])
   428→    let cur = a
   429→    for (const [g0, g1] of gaps) { put((cur + g0) / 2, g0 - cur, bandLo, bandHi); cur = g1 }
   430→    put((cur + b) / 2, b - cur, bandLo, bandHi)
   431→  }
   432→}

   438→  return [cx + lx * c - lz * s, cz + lx * s + lz * c]
   439→}

   811→/** ¿Choca el segmento p→q con alguna caja? (para línea de visión) */
   812→export function segmentBlocked(px: number, py: number, pz: number, qx: number, qy: number, qz: number, boxes: AABB[] = MAP_AABBS): boolean {
   813→  const dx = qx - px, dy = qy - py, dz = qz - pz
   814→  for (let i = 0; i < boxes.length; i++) {
   815→    const b = boxes[i]
   816→    let tmin = 0, tmax = 1
   817→    // X
   818→    if (Math.abs(dx) < 1e-9) {
   819→      if (px < b.minX || px > b.maxX) continue
   820→    } else {
   821→      let t1 = (b.minX - px) / dx, t2 = (b.maxX - px) / dx
   822→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   823→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   824→      if (tmin > tmax) continue
   825→    }
   826→    // Y
   827→    if (Math.abs(dy) < 1e-9) {
   828→      if (py < b.minY || py > b.maxY) continue
   829→    } else {
   830→      let t1 = (b.minY - py) / dy, t2 = (b.maxY - py) / dy
   831→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   832→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   833→      if (tmin > tmax) continue
   834→    }
   835→    // Z
   836→    if (Math.abs(dz) < 1e-9) {
   837→      if (pz < b.minZ || pz > b.maxZ) continue
   838→    } else {
   839→      let t1 = (b.minZ - pz) / dz, t2 = (b.maxZ - pz) / dz
   840→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   841→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   842→      if (tmin > tmax) continue
   843→    }
   844→    return true
   845→  }
   846→  return false
   847→}
