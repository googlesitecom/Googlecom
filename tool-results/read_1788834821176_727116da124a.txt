   214→  private buildMap(): void {
   215→    const texs = makeWorldTextures()
   216→
   217→    // suelo
   218→    const groundMat = new THREE.MeshLambertMaterial({ map: texs.sand })
   219→    groundMat.map!.repeat.set(22, 22)
   220→    const ground = new THREE.Mesh(new THREE.PlaneGeometry(110, 110), groundMat)
   221→    ground.rotation.x = -Math.PI / 2
   222→    ground.receiveShadow = true
   223→    this.scene.add(ground)
   224→    this.shootables.push(ground)
   225→
   226→    // cajas del mapa
   227→    for (const b of MAP_BOXES) {
   228→      let mesh: THREE.Mesh
   229→      if (b.mat === 'barrel') {
   230→        mesh = new THREE.Mesh(
   231→          new THREE.CylinderGeometry(0.36, 0.36, b.h, 12),
   232→          new THREE.MeshLambertMaterial({ map: texs.barrel }),
   233→        )
   234→      } else {
   235→        const tex = texs[b.mat].clone()
   236→        tex.needsUpdate = true
   237→        tex.repeat.set(Math.max(1, Math.round(Math.max(b.w, b.d) / 2.5)), Math.max(1, Math.round(b.h / 2.5)))
   238→        mesh = new THREE.Mesh(
   239→          new THREE.BoxGeometry(b.w, b.h, b.d),
   240→          new THREE.MeshLambertMaterial({ map: tex }),
   241→        )
   242→        mesh.castShadow = true
   243→        mesh.receiveShadow = true
   244→      }
   245→      mesh.position.set(b.x, b.y, b.z)
   246→      this.scene.add(mesh)
   247→      this.shootables.push(mesh)
   248→      this.mapMeshes.push(mesh)
   249→    }
   250→
   251→    // marcas de spawn (zonas de compra)
   252→    for (const [sp, color] of [[SPAWN_A, 0xf59e0b], [SPAWN_B, 0x22c55e]] as [number[], number][]) {
   253→      const ring = new THREE.Mesh(
   254→        new THREE.RingGeometry(GAME.BUY_RADIUS - 0.15, GAME.BUY_RADIUS, 40),
   255→        new THREE.MeshBasicMaterial({ color, transparent: true, opacity: 0.35, side: THREE.DoubleSide }),
   256→      )
   257→      ring.rotation.x = -Math.PI / 2
   258→      ring.position.set(sp[0], 0.03, sp[2])
   259→      this.scene.add(ring)
   260→    }
   261→  }

   321→  private onKeyDown = (e: KeyboardEvent): void => {
   322→    const s = useGame.getState()
   323→    if (e.code === 'Tab') {
   324→      e.preventDefault()
   325→      s.setHud({ scoreboardOpen: true })
   326→      return
   327→    }
   328→    if (s.buyOpen) {
   329→      if (e.code === 'KeyB' || e.code === 'Escape') {
   330→        e.preventDefault()
   331→        this.closeBuyMenu()
   332→      }
   333→      return
   334→    }
   335→    if (s.phase !== 'playing') return
   336→    this.keys.add(e.code)
   337→    switch (e.code) {
   338→      case 'KeyB': this.openBuyMenu(); break
   339→      case 'KeyR': this.startReload(); break
   340→      case 'KeyG': this.throwGrenade(); break
   341→      case 'KeyQ': this.switchTo(this.lastWeapon); break
   342→      case 'Digit1': {
   343→        const p = PRIMARY_PREF.find(w => this.owned.includes(w))
   344→        if (p) this.switchTo(p)
   345→        break
   346→      }
   347→      case 'Digit2': {
   348→        const p = SECONDARY_PREF.find(w => this.owned.includes(w))
   349→        if (p) this.switchTo(p)
   350→        break
   351→      }
   352→      case 'Digit3': this.switchTo('knife'); break
   353→    }
   354→  }

   380→  private onMouseMove = (e: MouseEvent): void => {
   381→    if (!this.locked || this.dead) return
   382→    const s = useGame.getState()
   383→    const zoomFactor = this.adsAmt > 0.05 ? Math.max(0.28, this.camera.fov / BASE_FOV) : 1
   384→    const sens = 0.0021 * s.settings.sens * (this.ads ? zoomFactor : 1) * (this.sprinting ? 1.12 : 1)
   385→    this.yaw -= e.movementX * sens
   386→    this.pitch -= e.movementY * sens
   387→    this.pitch = Math.max(-1.45, Math.min(1.45, this.pitch))
   388→    // sway del arma
   389→    this.swayX = Math.max(-1, Math.min(1, this.swayX - e.movementX * 0.0006))
   390→    this.swayY = Math.max(-1, Math.min(1, this.swayY - e.movementY * 0.0006))
   391→  }

   425→  private loop = (): void => {
   426→    if (this.disposed) return
   427→    this.raf = requestAnimationFrame(this.loop)
   428→    const dt = Math.min(0.05, this.clock.getDelta())
   429→    const t = performance.now() / 1000
   430→
   431→    const phase = useGame.getState().phase
   432→    const playing = phase === 'playing' || phase === 'dead'
   433→
   434→    if (playing) {
   435→      this.updateMovement(dt)
   436→      this.updateWeapon(dt, t)
   437→      this.updateShooting(t)
   438→    }
   439→    this.updateCamera(dt, t)
   440→    this.updateViewmodel(dt, t)
   441→
   442→    // remotos (interpolación)
   443→    const renderT = performance.now() - GAME.INTERP_DELAY
   444→    const states = this.remotes.update(dt, renderT, this.team, this.camera.position)
   445→    this.updateRemoteFootsteps(dt, states)
   446→
   447→    // granadas visibles
   448→    this.updateGrenadeViews(dt)
   449→
   450→    // efectos
   451→    this.effects.update(dt, this.camera)
   452→    this.trauma = Math.max(0, this.trauma - dt * 1.4)
   453→    this.hurtFlash = Math.max(0, this.hurtFlash - dt * 1.6)
   454→
   455→    // HUD canvas
   456→    this.drawOverlay(t)
   457→    if (performance.now() - this.minimapT > 100) {
   458→      this.minimapT = performance.now()
   459→      this.drawMinimap()
   460→    }
   461→
   462→    // scoreboard y HUD numérico
   463→    if (performance.now() - this.scoreboardT > 400) {
   464→      this.scoreboardT = performance.now()
   465→      useGame.getState().setHud({ scoreboard: states })
   466→    }
   467→
   468→    // envío de input al servidor
   469→    if (playing && performance.now() - this.lastInputSent > GAME.INPUT_RATE) {
   470→      this.lastInputSent = performance.now()
   471→      this.net.sendInput()
   472→    }
   473→
   474→    // FPS
   475→    this.fpsFrames++
   476→    if (t - this.fpsT > 1) {
   477→      useGame.getState().setHud({ fps: Math.round(this.fpsFrames / (t - this.fpsT)) })
   478→      this.fpsFrames = 0
   479→      this.fpsT = t
   480→    }
   481→
   482→    this.renderer.render(this.scene, this.camera)
   483→  }

   492→  private collides(x: number, y: number, z: number, h: number): boolean {
   493→    const minX = x - HALF_W, maxX = x + HALF_W
   494→    const minY = y, maxY = y + h
   495→    const minZ = z - HALF_W, maxZ = z + HALF_W
   496→    for (let i = 0; i < MAP_AABBS.length; i++) {
   497→      const b = MAP_AABBS[i]
   498→      if (maxX > b.minX && minX < b.maxX && maxY > b.minY && minY < b.maxY && maxZ > b.minZ && minZ < b.maxZ) {
   499→        return true
   500→      }
   501→    }
   502→    return false
   503→  }

   505→  private updateMovement(dt: number): void {
   506→    const s = useGame.getState()
   507→    const inputActive = s.phase === 'playing' && this.locked && !this.dead
   508→
   509→    const w = WEAPONS[this.weapon]
   510→    const wantCrouch = inputActive && (this.keys.has('ControlLeft') || this.keys.has('KeyC'))
   511→    const canStand = !this.collides(this.pos.x, this.pos.y, this.pos.z, 1.8)
   512→    this.crouching = wantCrouch || (!canStand && this.pos.y < 3)
   513→
   514→    const wantSprint = inputActive && this.keys.has('ShiftLeft') && !this.crouching && !this.ads
   515→    const movingFwd = this.keys.has('KeyW')
   516→
   517→    let speed = 4.6 * w.moveMult
   518→    this.sprinting = false
   519→    if (wantSprint && movingFwd && this.onGround) {
   520→      speed *= 1.45
   521→      this.sprinting = true
   522→    }
   523→    if (this.crouching) speed *= 0.5
   524→    if (this.adsAmt > 0.3) speed *= 0.65
   525→
   526→    // dirección de input
   527→    let ix = 0, iz = 0
   528→    if (inputActive) {
   529→      if (this.keys.has('KeyW')) iz += 1
   530→      if (this.keys.has('KeyS')) iz -= 1
   531→      if (this.keys.has('KeyA')) ix -= 1
   532→      if (this.keys.has('KeyD')) ix += 1
   533→    }
   534→    const len = Math.hypot(ix, iz)
   535→    if (len > 0) { ix /= len; iz /= len }
   536→
   537→    // base yaw
   538→    const cos = Math.cos(this.yaw), sin = Math.sin(this.yaw)
   539→    const dirX = ix * cos + iz * sin
   540→    const dirZ = -ix * sin + iz * cos
   541→
   542→    // aceleración / fricción
   543→    const accel = this.onGround ? 12 : 2.2
   544→    const fric = this.onGround ? 10 : 0.3
   545→    this.vel.x += (dirX * speed - this.vel.x) * Math.min(1, accel * dt)
   546→    this.vel.z += (dirZ * speed - this.vel.z) * Math.min(1, accel * dt)
   547→    if (len === 0 && this.onGround) {
   548→      const damp = Math.max(0, 1 - fric * dt)
   549→      this.vel.x *= damp
   550→      this.vel.z *= damp
   551→    }
   552→
   553→    // salto / gravedad
   554→    if (inputActive && this.keys.has('Space') && this.onGround && !this.crouching) {
   555→      this.vel.y = 5.6
   556→      this.onGround = false
   557→      this.audio.jump()
   558→    }
   559→    this.vel.y -= GAME.GRAVITY * dt
   560→
   561→    // --- resolución por ejes ---
   562→    const h = this.playerHeight()
   563→    // eje X
   564→    let nx = this.pos.x + this.vel.x * dt
   565→    if (this.collides(nx, this.pos.y, this.pos.z, h)) {
   566→      // intentar subir escalón
   567→      if (this.onGround && !this.collides(nx, this.pos.y + 0.58, this.pos.z, h)) {
   568→        this.pos.y += 0.58
   569→        this.pos.x = nx
   570→      } else {
   571→        this.vel.x = 0
   572→        nx = this.pos.x
   573→      }
   574→    } else {
   575→      this.pos.x = nx
   576→    }
   577→    // eje Z
   578→    let nz = this.pos.z + this.vel.z * dt
   579→    if (this.collides(this.pos.x, this.pos.y, nz, h)) {
   580→      if (this.onGround && !this.collides(this.pos.x, this.pos.y + 0.58, nz, h)) {
   581→        this.pos.y += 0.58
   582→        this.pos.z = nz
   583→      } else {
   584→        this.vel.z = 0
   585→        nz = this.pos.z
   586→      }
   587→    } else {
   588→      this.pos.z = nz
   589→    }
   590→    // eje Y
   591→    let ny = this.pos.y + this.vel.y * dt
   592→    if (this.vel.y <= 0) {
   593→      // buscar suelo
   594→      let groundY = 0
   595→      for (let i = 0; i < MAP_AABBS.length; i++) {
   596→        const b = MAP_AABBS[i]
   597→        if (this.pos.x + HALF_W > b.minX && this.pos.x - HALF_W < b.maxX &&
   598→            this.pos.z + HALF_W > b.minZ && this.pos.z - HALF_W < b.maxZ) {
   599→          if (b.maxY <= this.pos.y + 0.01 && b.maxY > groundY) groundY = b.maxY
   600→        }
   601→      }
   602→      if (ny <= groundY + 0.001) {
   603→        if (!this.onGround && this.vel.y < -6) this.audio.land()
   604→        ny = groundY
   605→        this.vel.y = 0
   606→        this.onGround = true
   607→      } else {
   608→        this.onGround = false
   609→      }
   610→    } else {
   611→      if (this.collides(this.pos.x, ny, this.pos.z, h)) {
   612→        this.vel.y = 0
   613→        ny = this.pos.y
   614→      } else {
   615→        this.onGround = false
   616→      }
   617→    }
   618→    this.pos.y = Math.max(0, ny)
   619→    this.pos.x = Math.max(-34.2, Math.min(34.2, this.pos.x))
   620→    this.pos.z = Math.max(-34.2, Math.min(34.2, this.pos.z))
   621→
   622→    // bob y pasos
   623→    const hSpeed = Math.hypot(this.vel.x, this.vel.z)
   624→    if (this.onGround && hSpeed > 0.5) {
   625→      this.bobT += dt * hSpeed * 1.6
   626→      this.stepT += dt
   627→      const cadence = hSpeed > 5.5 ? 0.31 : hSpeed > 2 ? 0.42 : 0.55
   628→      if (this.stepT > cadence) {
   629→        this.stepT = 0
   630→        this.audio.footstep(0, true)
   631→      }
   632→    } else {
   633→      this.bobT += dt * 0.4
   634→    }
   635→    this.sprintAmt += ((this.sprinting && hSpeed > 4 ? 1 : 0) - this.sprintAmt) * Math.min(1, dt * 8)
   636→
   637→    // zona de compra
   638→    const sp = this.team === 'A' ? SPAWN_A : SPAWN_B
   639→    const inZone = Math.hypot(this.pos.x - sp[0], this.pos.z - sp[2]) < GAME.BUY_RADIUS
   640→    if (inZone !== useGame.getState().buyZone) {
   641→      useGame.getState().setHud({ buyZone: inZone })
   642→    }
   643→
   644→    // parámetros calculados (no usados directamente)
   645→  }

   650→  private updateCamera(dt: number, t: number): void {
   651→    // altura de ojos
   652→    const targetEye = this.dead ? 0.4 : this.crouching ? EYE_CROUCH : EYE_STAND
   653→    const curEye = this.camera.position.y - this.pos.y
   654→    const eye = curEye + (targetEye - curEye) * Math.min(1, dt * 10)
   655→
   656→    // bob
   657→    const hSpeed = Math.hypot(this.vel.x, this.vel.z)
   658→    const bobAmp = this.onGround ? Math.min(1, hSpeed / 5) * (this.adsAmt > 0.3 ? 0.008 : 0.028) : 0
   659→    const bobX = Math.cos(this.bobT) * bobAmp * 0.6
   660→    const bobY = Math.abs(Math.sin(this.bobT)) * bobAmp
   661→
   662→    // sacudida (trauma)
   663→    const sh = this.trauma * this.trauma
   664→    const shakeP = (Math.random() - 0.5) * 0.05 * sh
   665→    const shakeR = (Math.random() - 0.5) * 0.04 * sh
   666→
   667→    this.camera.position.set(
   668→      this.pos.x + bobX,
   669→      this.pos.y + eye + bobY - (this.dead ? this.deathT * 1.2 : 0),
   670→      this.pos.z,
   671→    )
   672→
   673→    // FOV
   674→    const w = WEAPONS[this.weapon]
   675→    const targetFov = this.ads && !this.dead ? w.zoomFov : this.sprintAmt > 0.3 ? BASE_FOV + 7 * this.sprintAmt : BASE_FOV
   676→    this.camera.fov += (targetFov - this.camera.fov) * Math.min(1, dt * 10)
   677→    this.camera.updateProjectionMatrix()
   678→    this.adsAmt += ((this.ads && !this.dead && !this.reloading ? 1 : 0) - this.adsAmt) * Math.min(1, dt * 9)
   679→
   680→    // recuperación del retroceso
   681→    const rec = WEAPONS[this.weapon].recoilRecover
   682→    this.recoilP *= Math.max(0, 1 - rec * dt * 6)
   683→    this.recoilY *= Math.max(0, 1 - rec * dt * 6)
   684→    if (t * 1000 - this.lastShotTime > 380) {
   685→      this.sprayIdx = Math.max(0, this.sprayIdx - dt * 18)
   686→    }
   687→
   688→    // muerte: cámara cae
   689→    if (this.dead) this.deathT = Math.min(1, this.deathT + dt * 1.8)
   690→
   691→    this.camera.rotation.order = 'YXZ'
   692→    this.camera.rotation.y = this.yaw + this.recoilY + shakeR * 0.4
   693→    this.camera.rotation.x = Math.max(-1.5, Math.min(1.5, this.pitch + this.recoilP + shakeP - this.deathT * 0.8))
   694→    this.camera.rotation.z = shakeR + (this.dead ? this.deathT * 0.6 : 0) + Math.cos(this.bobT * 0.5) * bobAmp * 0.2
   695→  }

   753→  private updateViewmodel(dt: number, t: number): void {
   754→    if (!this.vmGroup) return
   755→    const pose = weaponPose(this.weapon)
   756→    this.drawT = Math.min(1, this.drawT + dt * 4.5)
   757→    const draw = this.drawT
   758→
   759→    // sway
   760→    this.swayX *= Math.max(0, 1 - dt * 6)
   761→    this.swayY *= Math.max(0, 1 - dt * 6)
   762→
   763→    // posiciones
   764→    const hip = pose.hip
   765→    const ads = pose.ads
   766→    const adsA = this.adsAmt
   767→    const sprintA = this.sprintAmt * (1 - adsA)
   768→    const vmKick = this.vmKick
   769→    this.vmKick *= Math.max(0, 1 - dt * 9)
   770→
   771→    // bob del arma
   772→    const hSpeed = Math.hypot(this.vel.x, this.vel.z)
   773→    const bob = this.onGround ? Math.min(1, hSpeed / 5) * (1 - adsA * 0.85) : 0
   774→
   775→    let px = hip.x + (ads.x - hip.x) * adsA
   776→    let py = hip.y + (ads.y - hip.y) * adsA
   777→    let pz = hip.z + (ads.z - hip.z) * adsA + vmKick * 0.09
   778→
   779→    // animación de recarga
   780→    let reloadRot = 0
   781→    if (this.reloading) {
   782→      const now = performance.now()
   783→      const w = WEAPONS[this.weapon]
   784→      const progress = 1 - (this.reloadEndAt - now) / (w.reloadTime * 1000)
   785→      const dip = Math.sin(Math.min(1, progress) * Math.PI)
   786→      reloadRot = dip * 0.9
   787→      py -= dip * 0.16
   788→      // sonidos por etapas
   789→      if (progress > 0.25 && this.reloadStage === 0) { this.reloadStage = 1; this.audio.reload('mag') }
   790→      if (progress > 0.85 && this.reloadStage === 1) { this.reloadStage = 2; this.audio.reload('end') }
   791→    }
   792→
   793→    px += Math.cos(this.bobT) * 0.012 * bob - this.swayX * 0.028 * (1 - adsA * 0.8)
   794→    py += Math.abs(Math.sin(this.bobT)) * 0.010 * bob + this.swayY * 0.024 * (1 - adsA * 0.8)
   795→    py -= (1 - draw) * 0.35   // animación de desenfundado
   796→    pz -= (1 - draw) * 0.12
   797→
   798→    this.vmGroup.position.set(px, py, pz)
   799→    this.vmGroup.rotation.set(
   800→      pose.hipRot.x + reloadRot + vmKick * 0.14 + this.swayY * 0.06 * (1 - adsA),
   801→      pose.hipRot.y * (1 - adsA) + sprintA * 0.5 - this.swayX * 0.05 * (1 - adsA),
   802→      pose.hipRot.z + sprintA * 0.25 + reloadRot * 0.4,
   803→    )
   804→    // sprint: arma apuntando abajo
   805→    if (sprintA > 0.01) {
   806→      this.vmGroup.rotation.x += sprintA * 0.5
   807→      this.vmGroup.position.y -= sprintA * 0.08
   808→    }
   809→
   810→    // francotirador ADS: ocultar modelo
   811→    const w = WEAPONS[this.weapon]
   812→    this.vmHolder.visible = !(w.sniper && adsA > 0.7) && !this.dead
   813→
   814→    void t
   815→  }

   825→  private currentSpread(): number {
   826→    const w = WEAPONS[this.weapon]
   827→    if (this.weapon === 'knife') return 0
   828→    let spread = w.spreadBase
   829→    const hSpeed = Math.hypot(this.vel.x, this.vel.z)
   830→    spread += w.spreadMove * Math.min(1, hSpeed / 6)
   831→    if (!this.onGround) spread += w.spreadAir
   832→    if (this.crouching) spread *= 0.72
   833→    if (this.adsAmt > 0.6) spread *= 0.45
   834→    spread += this.sprayIdx * w.sprayInacc
   835→    return spread
   836→  }

   838→  private updateShooting(t: number): void {
   839→    const s = useGame.getState()
   840→    if (s.phase !== 'playing' || this.dead || s.buyOpen) { this.shooting = false; return }
   841→    const w = WEAPONS[this.weapon]
   842→    const now = t * 1000
   843→
   844→    // fin de recarga
   845→    if (this.reloading && now >= this.reloadEndAt) {
   846→      this.reloading = false
   847→      const a = this.ammo[this.weapon]
   848→      if (a) {
   849→        const need = w.mag - a.mag
   850→        const take = Math.min(need, a.reserve)
   851→        a.mag += take
   852→        a.reserve -= take
   853→      }
   854→      this.updateHudWeapon()
   855→    }
   856→
   857→    if (!this.shooting || this.reloading || now < this.nextShotAt) return
   858→    if (this.drawT < 0.8) return
   859→
   860→    const a = this.ammo[this.weapon]
   861→    if (w.mag > 0 && (a?.mag ?? 0) <= 0) {
   862→      this.audio.dryFire()
   863→      this.nextShotAt = now + 250
   864→      this.shooting = false
   865→      this.startReload()
   866→      return
   867→    }
   868→
   869→    this.fireWeapon(t)
   870→  }

   872→  private fireWeapon(t: number): void {
   873→    const w = WEAPONS[this.weapon]
   874→    const now = t * 1000
   875→    this.nextShotAt = now + 60000 / w.rpm
   876→    const a = this.ammo[this.weapon]
   877→    if (a && w.mag > 0) a.mag--
   878→
   879→    // dirección con dispersión
   880→    this.camera.updateMatrixWorld()
   881→    const forward = new THREE.Vector3(0, 0, -1).applyQuaternion(this.camera.quaternion)
   882→    const right = new THREE.Vector3(1, 0, 0).applyQuaternion(this.camera.quaternion)
   883→    const up = new THREE.Vector3(0, 1, 0).applyQuaternion(this.camera.quaternion)
   884→
   885→    const eye = this.camera.position.clone()
   886→
   887→    if (this.weapon === 'knife') {
   888→      // ataque cuerpo a cuerpo
   889→      this.audio.knifeSwing()
   890→      this.vmKick = 1.4
   891→      const dir = forward.clone()
   892→      const hit = this.castBullet(eye, dir, 2.4)
   893→      if (hit) this.processHit(hit, dir)
   894→      this.updateHudWeapon()
   895→      return
   896→    }
   897→
   898→    const spread = this.currentSpread()
   899→    const pellets = w.pellets
   900→    const hits: { target: string; part: 'head' | 'body' | 'legs'; dist: number; point: THREE.Vector3 }[] = []
   901→
   902→    for (let i = 0; i < pellets; i++) {
   903→      const r = (spread * Math.PI / 180) * Math.sqrt(Math.random())
   904→      const ang = Math.random() * Math.PI * 2
   905→      const dir = forward.clone()
   906→        .addScaledVector(right, Math.cos(ang) * r)
   907→        .addScaledVector(up, Math.sin(ang) * r)
   908→        .normalize()
   909→
   910→      const hit = this.castBullet(eye, dir, 200)
   911→      if (!hit) {
   912→        // impacto lejano en el mapa
   913→        this.raycaster.set(eye, dir)
   914→        this.raycaster.far = 200
   915→        const intersect = this.raycaster.intersectObjects(this.shootables, false)[0]
   916→        if (intersect) {
   917→          const normal = intersect.face ? intersect.face.normal.clone().transformDirection(intersect.object.matrixWorld) : dir.clone().negate()
   918→          this.effects.impact(intersect.point, normal)
   919→          this.effects.tracer(this.muzzleWorld(), intersect.point)
   920→        }
   921→        continue
   922→      }
   923→      // efecto local
   924→      this.effects.tracer(this.muzzleWorld(), hit.point)
   925→      if (hit.player) {
   926→        this.effects.impact(hit.point, dir.clone().negate(), true)
   927→        hits.push({ target: hit.player, part: hit.part, dist: hit.dist, point: hit.point })
   928→      } else {
   929→        this.effects.impact(hit.point, hit.normal ?? dir.clone().negate())
   930→      }
   931→    }
   932→
   933→    // enviar aciertos al servidor (de una vez: válido para escopeta)
   934→    if (hits.length > 0) {
   935→      this.net.sendHits(this.weapon, hits.map(h => ({ target: h.target, part: h.part, dist: h.dist })))
   936→      if (this.audio) this.audio.fleshHit(0)
   937→    }
   938→
   939→    // audio + fogonazo
   940→    this.audio.gunshot(w.sound, 0)
   941→    const mzl = this.muzzleWorld()
   942→    this.effects.muzzleFlash(mzl, this.weapon === 'awp338' ? 1.6 : this.weapon === 'breacher' ? 1.3 : 1)
   943→    // casquillo
   944→    if (this.vmMuzzle) {
   945→      const ejectPos = mzl.clone().addScaledVector(forward, -0.25)
   946→      ejectPos.y += 0.05
   947→      const rightDir = right.clone()
   948→      this.effects.casing(ejectPos, rightDir)
   949→    }
   950→
   951→    // retroceso
   952→    const spray = this.sprayIdx
   953→    this.recoilP += w.recoilV * (0.85 + Math.random() * 0.3) * (this.crouching ? 0.85 : 1) * (this.adsAmt > 0.6 ? 0.8 : 1)
   954→    this.recoilY += w.recoilH * Math.sin(spray * 0.9 + 0.6) * (Math.random() * 0.5 + 0.75)
   955→    this.sprayIdx++
   956→    this.vmKick = 1
   957→    this.lastShotTime = now
   958→    this.trauma = Math.min(1, this.trauma + (w.id === 'awp338' ? 0.35 : w.id === 'breacher' ? 0.3 : 0.12))
   959→
   960→    // sonido de bomba para la escopeta
   961→    if (this.weapon === 'breacher') {
   962→      setTimeout(() => this.audio.reload('pump'), 260)
   963→    }
   964→    // recarga automática al vaciar
   965→    if (a && a.mag === 0 && a.reserve > 0) {
   966→      setTimeout(() => this.startReload(), 350)
   967→    }
   968→    this.updateHudWeapon()
   969→  }

   971→  private muzzleWorld(): THREE.Vector3 {
   972→    if (this.vmMuzzle) {
   973→      this.vmMuzzle.updateWorldMatrix(true, false)
   974→      return new THREE.Vector3().setFromMatrixPosition(this.vmMuzzle.matrixWorld)
   975→    }
   976→    return this.camera.position.clone()
   977→  }

   979→  /** Raycast local: mapa + hitboxes de jugadores remotos */
   980→  private castBullet(eye: THREE.Vector3, dir: THREE.Vector3, maxDist: number): {
   981→    player: string | null; part: 'head' | 'body' | 'legs'; dist: number; point: THREE.Vector3; normal?: THREE.Vector3
   982→  } | null {
   983→    // 1) mapa
   984→    this.raycaster.set(eye, dir)
   985→    this.raycaster.far = maxDist
   986→    const mapHits = this.raycaster.intersectObjects(this.shootables, false)
   987→    const mapHit = mapHits[0]
   988→    const mapDist = mapHit ? mapHit.distance : Infinity
   989→
   990→    // 2) jugadores remotos (enemigos vivos)
   991→    const ray = new THREE.Ray(eye, dir)
   992→    let bestPlayer: string | null = null
   993→    let bestPart: 'head' | 'body' | 'legs' = 'body'
   994→    let bestDist = Infinity
   995→    let bestPoint: THREE.Vector3 | null = null
   996→
   997→    for (const [id] of this.remotes.map) {
   998→      const st = this.remotes.map.get(id)!.state
   999→      if (!st || st.dead || st.team === this.team) continue
  1000→      const boxes = this.remotes.hitboxes(id)
  1001→      if (!boxes) continue
  1002→      const parts: ['head' | 'body' | 'legs', THREE.Box3][] = [
  1003→        ['head', boxes.head], ['body', boxes.body], ['legs', boxes.legs],
  1004→      ]
  1005→      for (const [part, box] of parts) {
  1006→        const p = new THREE.Vector3()
  1007→        if (ray.intersectBox(box, p)) {
  1008→          const d = p.distanceTo(eye)
  1009→          if (d < bestDist) {
  1010→            bestDist = d
  1011→            bestPlayer = id
  1012→            bestPart = part
  1013→            bestPoint = p
  1014→          }
  1015→        }
  1016→      }
  1017→    }
  1018→
  1019→    if (bestPlayer && bestDist < mapDist && bestPoint) {
  1020→      return { player: bestPlayer, part: bestPart, dist: bestDist, point: bestPoint }
  1021→    }
  1022→    if (mapHit) {
  1023→      const normal = mapHit.face ? mapHit.face.normal.clone().transformDirection(mapHit.object.matrixWorld) : dir.clone().negate()
