   193→/** Daño efectivo de un arma (sin escudo: la absorción la aplica la simulación) */
   194→export function computeDamage(w: WeaponConfig, part: BodyPart, distance: number): number {
   195→  let dmg = w.damage
   196→  if (part === 'head') dmg *= w.headMult
   197→  else if (part === 'legs') dmg *= w.legMult
   198→  if (distance > w.falloffStart) {
   199→    const t = Math.min(1, (distance - w.falloffStart) / Math.max(1, w.falloffEnd - w.falloffStart))
   200→    dmg *= 1 - (1 - w.falloffMin) * t
   201→  }
   202→  return Math.round(dmg)
   203→}

   277→/** Muro a lo largo del eje `ax`, en la coordenada fija `at`, desde `from` hasta `to`, con puerta centrada de ancho 2·gateHalf */
   278→function gatedWall(ax: 'x' | 'z', at: number, from: number, to: number, H: number, T: number, mat: MatKey, gateHalf = 1.6, doorH = 0) {
   279→  const lo = Math.min(from, to), hi = Math.max(from, to)
   280→  const mid = (lo + hi) / 2
   281→  const mk = (c: number, len: number) => {
   282→    if (len <= 0.05) return
   283→    if (ax === 'x') B(c, H / 2, at, len, H, T, mat)
   284→    else B(at, H / 2, c, T, H, len, mat)
   285→  }
   286→  mk((lo + (mid - gateHalf)) / 2, (mid - gateHalf) - lo)
   287→  mk((hi + (mid + gateHalf)) / 2, hi - (mid + gateHalf))
   288→  // dintel sobre la puerta (doorH = altura libre de la puerta)
   289→  if (gateHalf > 0 && H > doorH + 0.6) {
   290→    const lintelH = H - Math.max(doorH, 0.01)
   291→    if (lintelH > 0.15) {
   292→      if (ax === 'x') B(mid, H - lintelH / 2, at, gateHalf * 2, lintelH, T, mat)
   293→      else B(at, H - lintelH / 2, mid, T, lintelH, gateHalf * 2, mat)
   294→    }
   295→  }
   296→}

   552→/** ¿Choca el segmento p→q con alguna caja? (para línea de visión) */
   553→export function segmentBlocked(px: number, py: number, pz: number, qx: number, qy: number, qz: number, boxes: AABB[] = MAP_AABBS): boolean {
   554→  const dx = qx - px, dy = qy - py, dz = qz - pz
   555→  for (let i = 0; i < boxes.length; i++) {
   556→    const b = boxes[i]
   557→    let tmin = 0, tmax = 1
   558→    // X
   559→    if (Math.abs(dx) < 1e-9) {
   560→      if (px < b.minX || px > b.maxX) continue
   561→    } else {
   562→      let t1 = (b.minX - px) / dx, t2 = (b.maxX - px) / dx
   563→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   564→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   565→      if (tmin > tmax) continue
   566→    }
   567→    // Y
   568→    if (Math.abs(dy) < 1e-9) {
   569→      if (py < b.minY || py > b.maxY) continue
   570→    } else {
   571→      let t1 = (b.minY - py) / dy, t2 = (b.maxY - py) / dy
   572→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   573→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   574→      if (tmin > tmax) continue
   575→    }
   576→    // Z
   577→    if (Math.abs(dz) < 1e-9) {
   578→      if (pz < b.minZ || pz > b.maxZ) continue
   579→    } else {
   580→      let t1 = (b.minZ - pz) / dz, t2 = (b.maxZ - pz) / dz
   581→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   582→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   583→      if (tmin > tmax) continue
   584→    }
   585→    return true
   586→  }
   587→  return false
   588→}
