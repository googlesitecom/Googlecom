   236→/** Convierte un KeyboardEvent.code a etiqueta legible (KeyW → W, Digit1 → 1…) */
   237→export function keyLabel(code: string): string {
   238→  if (!code) return '—'
   239→  if (code.startsWith('Key')) return code.slice(3)
   240→  if (code.startsWith('Digit')) return code.slice(5)
   241→  if (code.startsWith('Numpad')) return 'NUM ' + code.slice(6)
   242→  if (code.startsWith('Arrow')) {
   243→    const names: Record<string, string> = { Up: '↑', Down: '↓', Left: '←', Right: '→' }
   244→    return names[code.slice(5)] ?? code
   245→  }
   246→  const names: Record<string, string> = {
   247→    Space: 'ESPACIO', ShiftLeft: 'MAYÚS IZQ', ShiftRight: 'MAYÚS DER',
   248→    ControlLeft: 'CTRL IZQ', ControlRight: 'CTRL DER',
   249→    AltLeft: 'ALT', AltRight: 'ALT GR', Enter: 'ENTER', Tab: 'TAB',
   250→    CapsLock: 'BLOQ MAYÚS', Backquote: '`', Minus: '-', Equal: '=',
   251→    BracketLeft: '[', BracketRight: ']', Semicolon: ';', Quote: "'",
   252→    Comma: ',', Period: '.', Slash: '/', Backslash: '\\',
   253→  }
   254→  return names[code] ?? code.toUpperCase()
   255→}

   266→/** Daño efectivo de un arma (sin escudo: la absorción la aplica la simulación) */
   267→export function computeDamage(w: WeaponConfig, part: BodyPart, distance: number): number {
   268→  let dmg = w.damage
   269→  if (part === 'head') dmg *= w.headMult
   270→  else if (part === 'legs') dmg *= w.legMult
   271→  if (distance > w.falloffStart) {
   272→    const t = Math.min(1, (distance - w.falloffStart) / Math.max(1, w.falloffEnd - w.falloffStart))
   273→    dmg *= 1 - (1 - w.falloffMin) * t
   274→  }
   275→  return Math.round(dmg)
   276→}

   349→// --- Muros con puerta central (edificios genéricos) ---
   350→/** Muro a lo largo del eje `ax`, en la coordenada fija `at`, desde `from` hasta `to`, con puerta centrada de ancho 2·gateHalf */
   351→function gatedWall(ax: 'x' | 'z', at: number, from: number, to: number, H: number, T: number, mat: MatKey, gateHalf = 1.6, doorH = 0) {
   352→  const lo = Math.min(from, to), hi = Math.max(from, to)
   353→  const mid = (lo + hi) / 2
   354→  const mk = (c: number, len: number) => {
   355→    if (len <= 0.05) return
   356→    if (ax === 'x') B(c, H / 2, at, len, H, T, mat)
   357→    else B(at, H / 2, c, T, H, len, mat)
   358→  }
   359→  mk((lo + (mid - gateHalf)) / 2, (mid - gateHalf) - lo)
   360→  mk((hi + (mid + gateHalf)) / 2, hi - (mid + gateHalf))
   361→  // dintel sobre la puerta (doorH = altura libre de la puerta)
   362→  if (gateHalf > 0 && H > doorH + 0.6) {
   363→    const lintelH = H - Math.max(doorH, 0.01)
   364→    if (lintelH > 0.15) {
   365→      if (ax === 'x') B(mid, H - lintelH / 2, at, gateHalf * 2, lintelH, T, mat)
   366→      else B(at, H - lintelH / 2, mid, T, lintelH, gateHalf * 2, mat)
   367→    }
   368→  }
   369→}

   371→// --- Mercado Central (0,0) 22×22, puertas de 3.2 m en cada lado ---
   372→const MW = 11, MH = 4.6, MT = 0.6, MG = 1.6
   373→for (const side of [-1, 1]) {
   374→  // Norte (z = -MW) y Sur (z = +MW)
   375→  const segW = MW - MG
   376→  B(-(MG + segW / 2), MH / 2, -MW, segW, MH, MT, 'sand')
   377→  B(MG + segW / 2, MH / 2, -MW, segW, MH, MT, 'sand')
   378→  B(-(MG + segW / 2), MH / 2, MW, segW, MH, MT, 'sand')

   657→/** ¿Choca el segmento p→q con alguna caja? (para línea de visión) */
   658→export function segmentBlocked(px: number, py: number, pz: number, qx: number, qy: number, qz: number, boxes: AABB[] = MAP_AABBS): boolean {
   659→  const dx = qx - px, dy = qy - py, dz = qz - pz
   660→  for (let i = 0; i < boxes.length; i++) {
   661→    const b = boxes[i]
   662→    let tmin = 0, tmax = 1
   663→    // X
   664→    if (Math.abs(dx) < 1e-9) {
   665→      if (px < b.minX || px > b.maxX) continue
   666→    } else {
   667→      let t1 = (b.minX - px) / dx, t2 = (b.maxX - px) / dx
   668→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   669→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   670→      if (tmin > tmax) continue
   671→    }
   672→    // Y
   673→    if (Math.abs(dy) < 1e-9) {
   674→      if (py < b.minY || py > b.maxY) continue
   675→    } else {
   676→      let t1 = (b.minY - py) / dy, t2 = (b.maxY - py) / dy
   677→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   678→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   679→      if (tmin > tmax) continue
   680→    }
   681→    // Z
   682→    if (Math.abs(dz) < 1e-9) {
   683→      if (pz < b.minZ || pz > b.maxZ) continue
   684→    } else {
   685→      let t1 = (b.minZ - pz) / dz, t2 = (b.maxZ - pz) / dz
   686→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   687→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   688→      if (tmin > tmax) continue
   689→    }
   690→    return true
   691→  }
   692→  return false
   693→}
