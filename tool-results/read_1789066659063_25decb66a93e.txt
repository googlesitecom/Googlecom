   293→  private spawnMarkers(objs: StoryObjective[], color: number): void {
   294→    this.beaconMat = new THREE.MeshBasicMaterial({ color, transparent: true, opacity: 0.5, depthWrite: false })

   298→  private spawnMarker(o: StoryObjective): void {
   299→    const group = new THREE.Group()
   300→    const beam = new THREE.Mesh(
   301→      new THREE.CylinderGeometry(0.5, 0.5, 26, 10, 1, true),
   302→      this.beaconMat,
   303→    )
   304→    beam.position.y = 13
   305→    group.add(beam)
   306→    const ring = new THREE.Mesh(
   307→      new THREE.RingGeometry(1.1, 1.5, 28),
   308→      new THREE.MeshBasicMaterial({ color: this.beaconMat.color.getHex(), transparent: true, opacity: 0.85, side: THREE.DoubleSide }),
   309→    )
   310→    ring.rotation.x = -Math.PI / 2
   311→    ring.position.y = 0.06
   312→    group.add(ring)
   313→    const label = this.makeLabel(o.label)
   314→    label.position.y = 2.6
   315→    group.add(label)
   316→    group.position.set(o.x, 0, o.z)
   317→    this.scene.add(group)
   318→    this.markers.push({ obj: o, group, beam, ring, label, done: false })
   319→  }

   321→  private makeLabel(text: string): THREE.Sprite {
   322→    const c = document.createElement('canvas')
   323→    c.width = 512
   324→    c.height = 128
   325→    const ctx = c.getContext('2d')!
   326→    ctx.fillStyle = 'rgba(10,12,10,0.82)'
   327→    ctx.fillRect(0, 34, 512, 60)
   328→    ctx.strokeStyle = 'rgba(217,160,91,0.9)'
   329→    ctx.lineWidth = 3
   330→    ctx.strokeRect(0, 34, 512, 60)
   331→    ctx.font = 'bold 40px "Arial Narrow", Arial, sans-serif'
   332→    ctx.fillStyle = '#ffe9c4'
   333→    ctx.textAlign = 'center'
   334→    ctx.textBaseline = 'middle'
   335→    ctx.fillText(text, 256, 66)
   336→    const tex = new THREE.CanvasTexture(c)
   337→    tex.colorSpace = THREE.SRGBColorSpace
   338→    const sprite = new THREE.Sprite(new THREE.SpriteMaterial({ map: tex, transparent: true, depthWrite: false }))
   339→    sprite.scale.set(7.5, 1.9, 1)
   340→    return sprite
   341→  }

   343→  private clearMarkers(): void {
   344→    for (const m of this.markers) {
   345→      this.scene.remove(m.group)
   346→      m.label.material.dispose()
   347→      ;(m.ring.material as THREE.Material).dispose()
   348→    }
   349→    this.markers.length = 0
   350→    this.beaconMat?.dispose()
   351→  }

   354→  minimapMarkers(): { x: number; z: number; color: string }[] {
   355→    const out: { x: number; z: number; color: string }[] = []
   356→    for (const m of this.markers) {

   358→      out.push({ x: m.obj.x, z: m.obj.z, color: '#ffb547' })

   363→  // ----------------------------------------------------------
   364→  // Diálogos de radio
   365→  // ----------------------------------------------------------
   366→  private nextDialogue(): void {
   367→    const lines = RADIO[this.chapter]
   368→    if (!lines) return
   369→    this.dialogueLine++
   370→    if (this.dialogueLine >= lines.length) { this.dialogueLine = lines.length - 1; return }
   371→    useGame.getState().setStory({ dialogue: lines[this.dialogueLine] })
   372→    this.dialogueT = 6.5
   373→  }

   375→  // ----------------------------------------------------------
   376→  // Bucle
   377→  // ----------------------------------------------------------
   378→  update(dt: number, t: number): void {
   379→    if (this.finished) return
   380→    const st = useGame.getState()
   381→    if (st.phase !== 'playing' && st.phase !== 'dead') return
   382→    // durante la cinemática el capítulo no avanza (el jugador no juega):
   383→    // invulnerable mientras no pueda moverse
   384→    // transición aplazada (inmune a la limitación de setTimeout)
   385→    if (this.scheduled && performance.now() >= this.scheduled.at) {
   386→      const fn = this.scheduled.fn
   387→      this.scheduled = null
   388→      fn()
   389→    }
   390→    if (st.cineActive) {
   391→      if (!this.cineProtected) {
   392→        this.cineProtected = true
   393→        this.game.net.sendStoryCmd({ cmd: 'protect', count: 22 })
   394→      }
   395→      return
   396→    }
   397→    this.cineProtected = false
   398→
   399→    // diálogo en pantalla
   400→    if (this.dialogueT > 0) {
   401→      this.dialogueT -= dt
   402→      if (this.dialogueT <= 0) useGame.getState().setStory({ dialogue: null })
   403→    }
   404→
   405→    // balizas: pulso
   406→    for (const m of this.markers) {
   407→      if (m.done) continue
   408→      const pulse = 0.35 + 0.3 * Math.sin(t * 3)
   409→      ;(m.beam.material as THREE.MeshBasicMaterial).opacity = pulse
   410→      const rs = 1 + 0.35 * Math.sin(t * 3)
   411→      m.ring.scale.set(rs, rs, 1)
   412→      m.label.position.y = 2.6 + 0.18 * Math.sin(t * 2)
   413→    }
   414→
   415→    // estadísticas
   416→    const kills = this.countKills()
   417→    const time = (performance.now() - this.startedAt) / 1000
   418→
   419→    if (!this.chapterLive) return
   420→
   421→    // ---- lógica por capítulo ----
   422→    if (this.chapter === 0) this.updateIntel(dt)
   423→    else if (this.chapter === 1) this.updateSiege(dt)
   424→    else if (this.chapter === 2) this.updateSabotage(dt)
   425→    else if (this.chapter === 3) this.updateRescue(dt)
   426→    else if (this.chapter === 4) this.updateBoss(dt)
   427→    else this.updateExtraction(dt)
   428→
   429→    // temporizador visible
   430→    const timer = Math.ceil(Math.max(0, this.timer))
   431→    useGame.getState().setStory({
   432→      timer,
   433→      stats: { time: Math.round(time), kills },
   434→    })
   435→  }

   437→  /** Cap 1: acercarse a las balizas las recoge */
   438→  private updateIntel(dt: number): void {
   439→    void dt
   440→    let done = 0
   441→    for (const m of this.markers) {
   442→      if (m.done) { done++; continue }
   443→      if (this.game.dead) break
   444→      const d = Math.hypot(this.game.pos.x - m.obj.x, this.game.pos.z - m.obj.z)
   445→      if (d < 2.6) {
   446→        m.done = true
   447→        this.scene.remove(m.group)
   448→        this.game.audio.pickup(false)
   449→        this.game.audio.announceDing()
   450→        done++
   451→        this.progress = `${done} / 3`
   452→        useGame.getState().setStory({ progress: this.progress })
   453→        this.nextDialogue()
   454→        // capítulo 1: entrega una MP-9 para el asedio

   464→  private updateSiege(dt: number): void {
   465→    const up = this.markers[0]
   466→    this.timer = Math.max(0, this.timer - dt)
   467→    const d = up ? Math.hypot(this.game.pos.x - up.obj.x, this.game.pos.z - up.obj.z) : 0
   468→    const far = d > 26
   469→    useGame.getState().setStory({
   470→      hint: far ? '¡VUELVE AL ENLACE! La subida se interrumpe si te alejas' : '',
   471→      timer: Math.ceil(this.timer),
   472→    })
   473→    // oleadas: los enemigos convergen sobre el enlace cada 22 s
   474→    if (t1000() - this.siegeWaveAt > 22000) {
   475→      this.siegeWaveAt = t1000()
   476→      this.game.net.sendStoryCmd({ cmd: 'attack', x: STORY_UPLINK.x, z: STORY_UPLINK.z })
   477→    }
   478→    if (far) return      // el temporizador no avanza lejos del enlace
   479→    const half = this.timer < 120
   480→    const quarter = this.timer < 60
   481→    if (half && !this.siegeHalf) {
   482→      this.siegeHalf = true
   483→      this.nextDialogue()
   484→      this.game.net.sendStoryCmd({ cmd: 'reinforce', count: 3 })
   485→    }
   486→    if (quarter && !this.siegeQuarter) {
   487→      this.siegeQuarter = true
   488→      this.nextDialogue()
   489→      this.game.net.sendStoryCmd({ cmd: 'reinforce', count: 2 })
   490→    }
   491→    if (this.timer <= 0 && !this.siegeDone) {
   492→      this.siegeDone = true
   493→      this.schedule(1.2, () => this.completeChapter())
   494→    }
   495→  }

   502→  /** Cap 3: mantener E junto a cada antena */
   503→  private updateSabotage(dt: number): void {
   504→    const code = this.game.bindCode('zipline') || 'KeyE'
   505→    const held = this.game.heldKeys().has(code)
   506→    let done = 0
   507→    let near: Marker | null = null
   508→    for (const m of this.markers) {
   509→      if (m.done) { done++; continue }
   510→      const d = Math.hypot(this.game.pos.x - m.obj.x, this.game.pos.z - m.obj.z)
   511→      if (d < 3.2) near = m
   512→    }
   513→    if (near) {
   514→      if (held && !this.game.dead) {
   515→        this.holdT += dt
   516→        this.holdTarget = near
   517→        if (this.holdT >= 2.5) {
   518→          near.done = true
   519→          this.scene.remove(near.group)
   520→          this.holdT = 0
   521→          this.holdTarget = null
   522→          done++
   523→          this.game.audio.reload('pump')
   524→          this.game.audio.announceDing()
   525→          this.nextDialogue()
   526→          useGame.getState().setStory({ progress: `${done} / 3` })
   527→          if (done === 3) {
   528→            // cuenta atrás de explosión
   529→            this.timer = 45
   530→            this.objective = '¡APÁRTATE! Cargas activas'
   531→            useGame.getState().setStory({ objective: this.objective })
   532→          }
   533→        }
   534→      } else if (this.holdTarget === near) {
   535→        this.holdT = Math.max(0, this.holdT - dt * 3)
   536→      }
   537→      const pct = Math.round((this.holdT / 2.5) * 100)
   538→      useGame.getState().setStory({
   539→        hint: this.holdT > 0.05 ? `COLOCANDO CARGA… ${pct}%` : 'MANTÉN [E] PARA COLOCAR LA CARGA',
   540→      })
   541→    } else {
   542→      this.holdT = 0
   543→      this.holdTarget = null
   544→      useGame.getState().setStory({ hint: '' })
   545→    }
   546→    // fase de explosión: cuenta atrás y ¡bum! (una sola vez)
   547→    if (this.timer > 0 && !this.boomDone) {
   548→      this.timer = Math.max(0, this.timer - dt)
   549→      useGame.getState().setStory({ timer: Math.ceil(this.timer) })
   550→      if (this.timer <= 0) {
   551→        this.boomDone = true
   552→        for (const a of STORY_ANTENNAS) this.game.onGrenadeExplode([a.x, 1.5, a.z])
   553→        this.game.audio.explosion(0)
   554→        this.game.audio.roundEnd()
   555→        this.schedule(1.6, () => this.completeChapter())
   556→      }
   557→    }
   558→  }

   566→  private updateRescue(dt: number): void {
   567→    const code = this.game.bindCode('zipline') || 'KeyE'
   568→    const held = this.game.heldKeys().has(code)
   569→
   570→    if (!this.rescueFreed) {
   571→      const m = this.markers[0]
   572→      if (!m || m.done) return
   573→      const d = Math.hypot(this.game.pos.x - m.obj.x, this.game.pos.z - m.obj.z)
   574→      if (d < 3.4) {
   575→        if (held && !this.game.dead) {
   576→          this.holdT += dt
   577→          if (this.holdT >= 3.0) {
   578→            this.rescueFreed = true
   579→            m.done = true
   580→            this.scene.remove(m.group)
   581→            this.holdT = 0
   582→            this.timer = 75          // alarma: sobrevivir mientras Ríos huye
   583→            this.nextDialogue()
   584→            this.objective = 'SOBREVIVE la alarma mientras Ríos escapa'
   585→            useGame.getState().setStory({ objective: this.objective, progress: 'ALERTA MÁXIMA', timer: 75 })
   586→            this.game.audio.announceDing()
   587→            this.game.net.sendStoryCmd({ cmd: 'reinforce', count: 4 })
   588→          }
   589→        } else {
   590→          this.holdT = Math.max(0, this.holdT - dt * 3)
   591→        }
   592→        const pct = Math.round((this.holdT / 3.0) * 100)
   593→        useGame.getState().setStory({
   594→          hint: this.holdT > 0.05 ? `ABRIENDO LA CELDA… ${pct}%` : 'MANTÉN [E] PARA LIBERAR AL PRISIONERO',
   595→        })
   596→      } else {
   597→        this.holdT = 0
   598→        useGame.getState().setStory({
   599→          hint: `PRISIONERO A ${Math.round(d)} m · alcanza la celda y mantén [E]`,
   600→        })
   601→      }
   602→      return
   603→    }
   604→
   605→    // fase de alarma: cronómetro con refuerzo final y presión constante
   606→    this.timer = Math.max(0, this.timer - dt)
   607→    useGame.getState().setStory({ timer: Math.ceil(this.timer) })
   608→    if (t1000() - this.rescueWaveAt > 25000) {
   609→      this.rescueWaveAt = t1000()
   610→      this.game.net.sendStoryCmd({ cmd: 'attack', x: STORY_PRISONER.x, z: STORY_PRISONER.z })
   611→    }
   612→    if (this.timer < 30 && !this.rescueReinforced) {
   613→      this.rescueReinforced = true
   614→      this.game.net.sendStoryCmd({ cmd: 'reinforce', count: 2 })
   615→    }
   616→    if (this.timer <= 0 && !this.rescueDone) {
   617→      this.rescueDone = true
   618→      useGame.getState().setStory({ hint: '', progress: '' })
   619→      this.schedule(1.2, () => this.completeChapter())
   620→    }
   621→  }

   666→    if (d < 4.5 && !this.finished) {
   667→      this.finished = true
   668→      this.nextDialogue()
   669→      const time = Math.round((performance.now() - this.startedAt) / 1000)
   670→      useGame.getState().setStory({
   671→        hint: '',
   672→        timer: 0,
   673→        stats: { time, kills: this.countKills() },
   674→      })
   675→      this.game.audio.roundEnd()
   676→      // cinemática final: órbita del helipuerto → pantalla de victoria

   681→        dur: c.dur,
