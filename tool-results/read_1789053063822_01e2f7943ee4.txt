   314→/** Índice del botón del ratón (Mouse2 → 2), o null */
   315→export function mouseButtonIndex(code: string): number | null {

   318→  return Number.isFinite(n) ? n : null
   319→}

   321→/** Convierte un KeyboardEvent.code o botón a etiqueta legible */
   322→export function keyLabel(code: string): string {
   323→  if (!code) return '—'
   324→  if (code.startsWith('Mouse')) {
   325→    const names = ['CLIC IZQ', 'CLIC MED', 'CLIC DER', 'CLIC 4', 'CLIC 5', 'CLIC 6']
   326→    const n = Number(code.slice(5))
   327→    return names[n] ?? `CLIC ${n + 1}`
   328→  }
   329→  if (code.startsWith('Key')) return code.slice(3)
   330→  if (code.startsWith('Digit')) return code.slice(5)
   331→  if (code.startsWith('Numpad')) return 'NUM ' + code.slice(6)
   332→  if (code.startsWith('Arrow')) {
   333→    const names: Record<string, string> = { Up: '↑', Down: '↓', Left: '←', Right: '→' }
   334→    return names[code.slice(5)] ?? code
   335→  }
   336→  const names: Record<string, string> = {
   337→    Space: 'ESPACIO', ShiftLeft: 'MAYÚS IZQ', ShiftRight: 'MAYÚS DER',
   338→    ControlLeft: 'CTRL IZQ', ControlRight: 'CTRL DER',
   339→    AltLeft: 'ALT', AltRight: 'ALT GR', Enter: 'ENTER', Tab: 'TAB',
   340→    CapsLock: 'BLOQ MAYÚS', Backquote: '`', Minus: '-', Equal: '=',
   341→    BracketLeft: '[', BracketRight: ']', Semicolon: ';', Quote: "'",
   342→    Comma: ',', Period: '.', Slash: '/', Backslash: '\\',
   343→  }
   344→  return names[code] ?? code.toUpperCase()
   345→}

   405→/** Daño efectivo de un arma (sin escudo: la absorción la aplica la simulación) */
   406→export function computeDamage(w: WeaponConfig, part: BodyPart, distance: number): number {
   407→  let dmg = w.damage
   408→  if (part === 'head') dmg *= w.headMult
   409→  else if (part === 'legs') dmg *= w.legMult
   410→  if (distance > w.falloffStart) {
   411→    const t = Math.min(1, (distance - w.falloffStart) / Math.max(1, w.falloffEnd - w.falloffStart))
   412→    dmg *= 1 - (1 - w.falloffMin) * t
   413→  }
   414→  return Math.round(dmg)
   415→}

   493→function gatedWall(ax: 'x' | 'z', at: number, from: number, to: number, H: number, T: number, mat: MatKey, gateHalf = 1.6, doorH = 0): void {
   494→  const lo = Math.min(from, to), hi = Math.max(from, to)
   495→  const mid = (lo + hi) / 2
   496→  const mk = (c: number, len: number) => {
   497→    if (len <= 0.05) return
   498→    if (ax === 'x') B(c, H / 2, at, len, H, T, mat)
   499→    else B(at, H / 2, c, T, H, len, mat)
   500→  }
   501→  mk((lo + (mid - gateHalf)) / 2, (mid - gateHalf) - lo)
   502→  mk((hi + (mid + gateHalf)) / 2, hi - (mid + gateHalf))
   503→  if (gateHalf > 0 && H > doorH + 0.6) {
   504→    const lintelH = H - Math.max(doorH, 0.01)
   505→    if (lintelH > 0.15) {
   506→      if (ax === 'x') B(mid, H - lintelH / 2, at, gateHalf * 2, lintelH, T, mat)
   507→      else B(at, H - lintelH / 2, mid, T, lintelH, gateHalf * 2, mat)
   508→    }
   509→  }
   510→}

   515→const ROT = { N: 0, E: Math.PI / 2, S: Math.PI, W: Math.PI * 1.5 } as const
   516→type Facing = keyof typeof ROT
   517→
   518→/** Caja en coordenadas locales del edificio, girada según `f` */
   519→function BR(cx: number, cz: number, f: Facing, lx: number, ly: number, lz: number, w: number, h: number, d: number, mat: MatKey): void {
   520→  const ang = ROT[f]
   521→  const c = Math.cos(ang), s = Math.sin(ang)
   522→  const rx = lx * c - lz * s
   523→  const rz = lx * s + lz * c
   524→  const swap = Math.abs(s) > 0.5
   525→  B(cx + rx, ly, cz + rz, swap ? d : w, h, swap ? w : d, mat)
   526→}

   528→interface WallOpts {
   529→  H?: number; T?: number; y0?: number
   530→  door?: number; doorHalf?: number; doorH?: number
   531→  wins?: number[]; winW?: number; bandLo?: number; bandHi?: number
   532→}
   533→
   534→/**
   535→ * Muro de edificio en coords. locales. `axis 'z'`: corre a lo largo de x en z=at;
   536→ * `axis 'x'`: corre a lo largo de z en x=at. Puerta = hueco a toda altura con dintel;
   537→ * ventanas = banda practiable (se puede disparar a través) entre zócalo y franja superior.
   538→ */
   539→function wallL(cx: number, cz: number, f: Facing, axis: 'x' | 'z', at: number, from: number, to: number, mat: MatKey, opts: WallOpts): void {
   540→  const H = opts.H ?? 3.3
   541→  const T = opts.T ?? 0.35
   542→  const y0 = opts.y0 ?? 0
   543→  const bandLo = y0 + (opts.bandLo ?? 1.45)
   544→  const bandHi = y0 + (opts.bandHi ?? 2.25)
   545→  const winW = opts.winW ?? 1.3
   546→  const lo = Math.min(from, to), hi = Math.max(from, to)
   547→  const put = (c: number, len: number, ya: number, yb: number) => {
   548→    if (len <= 0.06 || yb - ya <= 0.06) return
   549→    const cy = (ya + yb) / 2
   550→    if (axis === 'z') BR(cx, cz, f, c, cy, at, len, yb - ya, T, mat)
   551→    else BR(cx, cz, f, at, cy, c, T, yb - ya, len, mat)
   552→  }
   553→  let solids: [number, number][] = [[lo, hi]]
   554→  if (opts.door !== undefined) {
   555→    const dh = opts.doorHalf ?? 1.05
   556→    const doorH = y0 + (opts.doorH ?? 2.15)
   557→    const dLo = opts.door - dh, dHi = opts.door + dh
   558→    solids = []
   559→    if (dLo - lo > 0.05) solids.push([lo, dLo])
   560→    if (hi - dHi > 0.05) solids.push([dHi, hi])
   561→    put(opts.door, dh * 2, doorH, y0 + H)
   562→  }
   563→  for (const [a, b] of solids) {
   564→    put((a + b) / 2, b - a, y0, bandLo)
   565→    put((a + b) / 2, b - a, bandHi, y0 + H)
   566→    const gaps: [number, number][] = []
   567→    for (const w of opts.wins ?? []) {
   568→      const g0 = w - winW / 2, g1 = w + winW / 2
   569→      if (g0 > a + 0.3 && g1 < b - 0.3) gaps.push([g0, g1])
   570→    }
   571→    gaps.sort((p, q) => p[0] - q[0])
   572→    let cur = a
   573→    for (const [g0, g1] of gaps) { put((cur + g0) / 2, g0 - cur, bandLo, bandHi); cur = g1 }
   574→    put((cur + b) / 2, b - cur, bandLo, bandHi)
   575→  }
   576→}

   579→function wpTransform(cx: number, cz: number, f: Facing, lx: number, lz: number): [number, number] {
   580→  const ang = ROT[f], c = Math.cos(ang), s = Math.sin(ang)
   581→  return [cx + lx * c - lz * s, cz + lx * s + lz * c]
   582→}

   586→function stairsBR(cx: number, cz: number, f: Facing, xAt: number, zAt: number, dx: number, dz: number, steps: number, riseStep: number, runStep: number, wStair: number, y0 = 0, mat: MatKey = 'concrete'): void {

   592→    BR(cx, cz, f, sx, y0 + rise / 2, sz, dx !== 0 ? runStep + 0.04 : wStair, rise, dz !== 0 ? runStep + 0.04 : wStair, mat)

  1253→/** ¿Choca el segmento p→q con alguna caja? (para línea de visión) */
  1254→export function segmentBlocked(px: number, py: number, pz: number, qx: number, qy: number, qz: number, boxes: AABB[] = MAP_AABBS): boolean {
  1255→  const dx = qx - px, dy = qy - py, dz = qz - pz
  1256→  for (let i = 0; i < boxes.length; i++) {
  1257→    const b = boxes[i]
  1258→    let tmin = 0, tmax = 1
  1259→    // X
  1260→    if (Math.abs(dx) < 1e-9) {
  1261→      if (px < b.minX || px > b.maxX) continue
  1262→    } else {
  1263→      let t1 = (b.minX - px) / dx, t2 = (b.maxX - px) / dx
  1264→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
  1265→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
  1266→      if (tmin > tmax) continue
  1267→    }
  1268→    // Y
  1269→    if (Math.abs(dy) < 1e-9) {
  1270→      if (py < b.minY || py > b.maxY) continue
  1271→    } else {
  1272→      let t1 = (b.minY - py) / dy, t2 = (b.maxY - py) / dy
  1273→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
  1274→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
  1275→      if (tmin > tmax) continue
  1276→    }
  1277→    // Z
  1278→    if (Math.abs(dz) < 1e-9) {
  1279→      if (pz < b.minZ || pz > b.maxZ) continue
  1280→    } else {
  1281→      let t1 = (b.minZ - pz) / dz, t2 = (b.maxZ - pz) / dz
  1282→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
  1283→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
  1284→      if (tmin > tmax) continue
  1285→    }
  1286→    return true
  1287→  }
  1288→  return false
  1289→}

  1475→/** muro que corre a lo largo de X (en z=at), con puerta opcional en x=door */
  1476→function SWX(z: number, x0: number, x1: number, H: number, mat: MatKey, door?: number, dh = 1.15): void {
  1477→  const lo = Math.min(x0, x1), hi = Math.max(x0, x1)
  1478→  if (door === undefined) { SB((lo + hi) / 2, H / 2, z, hi - lo, H, 0.35, mat); return }
  1479→  if (door - dh - lo > 0.05) SB((lo + door - dh) / 2, H / 2, z, door - dh - lo, H, 0.35, mat)
  1480→  if (hi - door - dh > 0.05) SB((hi + door + dh) / 2, H / 2, z, hi - door - dh, H, 0.35, mat)
  1481→  const lint = H - 2.25
  1482→  if (lint > 0.15) SB(door, 2.25 + lint / 2, z, dh * 2, lint, 0.35, mat)
  1483→}
  1484→/** muro que corre a lo largo de Z (en x=at), con puerta opcional en z=door */
  1485→function SWZ(x: number, z0: number, z1: number, H: number, mat: MatKey, door?: number, dh = 1.15): void {
  1486→  const lo = Math.min(z0, z1), hi = Math.max(z0, z1)
  1487→  if (door === undefined) { SB(x, H / 2, (lo + hi) / 2, 0.35, H, hi - lo, mat); return }
  1488→  if (door - dh - lo > 0.05) SB(x, H / 2, (lo + door - dh) / 2, 0.35, H, door - dh - lo, mat)
  1489→  if (hi - door - dh > 0.05) SB(x, H / 2, (hi + door + dh) / 2, 0.35, H, hi - door - dh, mat)
  1490→  const lint = H - 2.25
  1491→  if (lint > 0.15) SB(x, 2.25 + lint / 2, door, 0.35, lint, dh * 2, mat)
  1492→}
  1493→/** escalera recta: peldaño i con altura creciente, avanza (dx,dz) por peldaño */
  1494→function SST(xAt: number, zAt: number, dx: number, dz: number, steps: number, rise: number, run: number, w: number, y0 = 0): void {

  1498→  }
