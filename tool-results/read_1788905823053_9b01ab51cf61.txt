   188→// Daño efectivo (usado por el servidor)
   189→export function computeDamage(w: WeaponConfig, part: BodyPart, distance: number, victimArmor: number): number {
   190→  let dmg = w.damage
   191→  if (part === 'head') dmg *= w.headMult
   192→  else if (part === 'legs') dmg *= w.legMult
   193→  // caída por distancia
   194→  if (distance > w.falloffStart) {
   195→    const t = Math.min(1, (distance - w.falloffStart) / Math.max(1, w.falloffEnd - w.falloffStart))
   196→    dmg *= 1 - (1 - w.falloffMin) * t
   197→  }
   198→  // armadura (la cabeza sin casco ya multiplicó; simplificamos: armadura protege cuerpo y cabeza)
   199→  if (victimArmor > 0 && part !== 'legs') {
   200→    dmg *= 1 - w.armorAbsorb
   201→  }
   202→  return Math.round(dmg)
   203→}

   271→// --- Cabañas (8x8, h3) en las cuatro diagonales medias ---
   272→function hut(cx: number, cz: number, doorSide: 'n' | 's' | 'e' | 'w', door2?: 'n' | 's' | 'e' | 'w') {
   273→  const S = 4, H = 3, T2 = 0.4, G = 1.5
   274→  const sides: ('n' | 's' | 'e' | 'w')[] = ['n', 's', 'e', 'w']
   275→  for (const s of sides) {
   276→    const hasDoor = s === doorSide || s === door2
   277→    const segLen = hasDoor ? (S - G) : (2 * S)
   278→    if (s === 'n') {
   279→      if (hasDoor) {
   280→        B(cx - (G + segLen / 2), H / 2, cz - S, segLen, H, T2, 'sand')
   281→        B(cx + (G + segLen / 2), H / 2, cz - S, segLen, H, T2, 'sand')
   282→        B(cx, H - 0.4, cz - S, G * 2, 0.8, T2, 'sand')
   283→      } else B(cx, H / 2, cz - S, 2 * S, H, T2, 'sand')
   284→    }
   285→    if (s === 's') {
   286→      if (hasDoor) {
   287→        B(cx - (G + segLen / 2), H / 2, cz + S, segLen, H, T2, 'sand')
   288→        B(cx + (G + segLen / 2), H / 2, cz + S, segLen, H, T2, 'sand')
   289→        B(cx, H - 0.4, cz + S, G * 2, 0.8, T2, 'sand')
   290→      } else B(cx, H / 2, cz + S, 2 * S, H, T2, 'sand')
   291→    }
   292→    if (s === 'e') {
   293→      if (hasDoor) {
   294→        B(cx + S, H / 2, cz - (G + segLen / 2), T2, H, segLen, 'sand')
   295→        B(cx + S, H / 2, cz + (G + segLen / 2), T2, H, segLen, 'sand')
   296→        B(cx + S, H - 0.4, cz, T2, 0.8, G * 2, 'sand')
   297→      } else B(cx + S, H / 2, cz, T2, H, 2 * S, 'sand')
   298→    }
   299→    if (s === 'w') {
   300→      if (hasDoor) {
   301→        B(cx - S, H / 2, cz - (G + segLen / 2), T2, H, segLen, 'sand')
   302→        B(cx - S, H / 2, cz + (G + segLen / 2), T2, H, segLen, 'sand')
   303→        B(cx - S, H - 0.4, cz, T2, 0.8, G * 2, 'sand')
   304→      } else B(cx - S, H / 2, cz, T2, H, 2 * S, 'sand')
   305→    }
   306→  }
   307→  B(cx, H + 0.15, cz, 9, 0.3, 9, 'roof')
   308→  B(cx + 1.2, 0.6, cz - 1.2, 1.2, 1.2, 1.2, 'crate')
   309→  B(cx - 1.5, 0.45, cz + 1.5, 0.7, 0.9, 0.7, 'barrel')
   310→}
   311→hut(26, 26, 'w')
   312→hut(-26, -26, 'e')
   313→hut(26, -26, 'w')
   314→hut(-26, 26, 'e')

   316→// --- Torres de vigilancia (0,±20), escaleras hacia el centro ---
   317→function tower(cz: number, stairsFrom: number): void {
   318→  B(0, 1.6, cz, 2.0, 3.2, 2.0, 'concrete')            // base/columna
   319→  B(0, 3.5, cz, 4.4, 0.3, 4.4, 'concrete')            // plataforma
   320→  B(0, 3.9, cz - 2.1, 4.4, 0.55, 0.3, 'metalOrange')  // baranda norte
   321→  B(0, 3.9, cz + 2.1, 4.4, 0.55, 0.3, 'metalOrange')  // baranda sur
   322→  B(-2.1, 3.9, cz, 0.3, 0.55, 4.4, 'metalOrange')     // baranda oeste
   323→  B(2.1, 3.9, cz, 0.3, 0.55, 4.4, 'metalOrange')      // baranda este
   324→  // escalera: sube hacia la plataforma (escalones de 0.5 m)
   325→  for (let i = 0; i < 7; i++) {
   326→    const h = 0.5 * (i + 1)
   327→    const z = cz + stairsFrom * (6.0 - i * 0.62)
   328→    B(0, h / 2 - 0.02, z, 1.6, h, 0.62, 'concrete')
   329→  }
   330→}
   331→tower(-20, 1)   // torre norte, escaleras al sur
   332→

   335→// --- Almacenes (0,±36) 22×15 con puertas frontal y laterales ---
   336→function warehouse(cz: number, faceTo: number): void {
   337→  const XW = 11, ZW = 7.5, H = 5, TW = 0.55, G = 2.2
   338→  // pared trasera
   339→  B(0, H / 2, cz - faceTo * ZW, XW * 2, H, TW, 'metalBlue')
   340→  // pared frontal con puerta central
   341→  const fz = cz + faceTo * ZW
   342→  const seg = XW - G
   343→  B(-(G + seg / 2), H / 2, fz, seg, H, TW, 'metalBlue')
   344→  B(G + seg / 2, H / 2, fz, seg, H, TW, 'metalBlue')
   345→  B(0, H - 0.5, fz, G * 2, 1, TW, 'metalBlue')
   346→  // laterales con puertas
   347→  for (const sx of [-1, 1]) {

   481→export function boxToAABB(b: MapBox): AABB {
   482→  return {
   483→    minX: b.x - b.w / 2, maxX: b.x + b.w / 2,
   484→    minY: b.y - b.h / 2, maxY: b.y + b.h / 2,
   485→    minZ: b.z - b.d / 2, maxZ: b.z + b.d / 2,
   486→  }
   487→}

   491→/** ¿Choca el segmento p→q con alguna caja? (para línea de visión) */
   492→export function segmentBlocked(px: number, py: number, pz: number, qx: number, qy: number, qz: number, boxes: AABB[] = MAP_AABBS): boolean {
   493→  const dx = qx - px, dy = qy - py, dz = qz - pz
   494→  for (let i = 0; i < boxes.length; i++) {
   495→    const b = boxes[i]
   496→    let tmin = 0, tmax = 1
   497→    // X
   498→    if (Math.abs(dx) < 1e-9) {
   499→      if (px < b.minX || px > b.maxX) continue
   500→    } else {
   501→      let t1 = (b.minX - px) / dx, t2 = (b.maxX - px) / dx
   502→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   503→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   504→      if (tmin > tmax) continue
   505→    }
   506→    // Y
   507→    if (Math.abs(dy) < 1e-9) {
   508→      if (py < b.minY || py > b.maxY) continue
   509→    } else {
   510→      let t1 = (b.minY - py) / dy, t2 = (b.maxY - py) / dy
   511→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   512→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   513→      if (tmin > tmax) continue
   514→    }
   515→    // Z
   516→    if (Math.abs(dz) < 1e-9) {
   517→      if (pz < b.minZ || pz > b.maxZ) continue
   518→    } else {
   519→      let t1 = (b.minZ - pz) / dz, t2 = (b.maxZ - pz) / dz
   520→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
   521→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
   522→      if (tmin > tmax) continue
   523→    }
   524→    return true
   525→  }
   526→  return false
   527→}
