   316→/** Convierte un KeyboardEvent.code o botón a etiqueta legible */
   317→export function keyLabel(code: string): string {
   318→  if (!code) return '—'
   319→  if (code.startsWith('Mouse')) {
   320→    const names = ['CLIC IZQ', 'CLIC MED', 'CLIC DER', 'CLIC 4', 'CLIC 5', 'CLIC 6']
   321→    const n = Number(code.slice(5))
   322→    return names[n] ?? `CLIC ${n + 1}`
   323→  }
   324→  if (code.startsWith('Key')) return code.slice(3)
   325→  if (code.startsWith('Digit')) return code.slice(5)
   326→  if (code.startsWith('Numpad')) return 'NUM ' + code.slice(6)
   327→  if (code.startsWith('Arrow')) {
   328→    const names: Record<string, string> = { Up: '↑', Down: '↓', Left: '←', Right: '→' }
   329→    return names[code.slice(5)] ?? code
   330→  }
   331→  const names: Record<string, string> = {
   332→    Space: 'ESPACIO', ShiftLeft: 'MAYÚS IZQ', ShiftRight: 'MAYÚS DER',
   333→    ControlLeft: 'CTRL IZQ', ControlRight: 'CTRL DER',
   334→    AltLeft: 'ALT', AltRight: 'ALT GR', Enter: 'ENTER', Tab: 'TAB',
   335→    CapsLock: 'BLOQ MAYÚS', Backquote: '`', Minus: '-', Equal: '=',
   336→    BracketLeft: '[', BracketRight: ']', Semicolon: ';', Quote: "'",
   337→    Comma: ',', Period: '.', Slash: '/', Backslash: '\\',
   338→  }
   339→  return names[code] ?? code.toUpperCase()
   340→}

   400→/** Daño efectivo de un arma (sin escudo: la absorción la aplica la simulación) */
   401→export function computeDamage(w: WeaponConfig, part: BodyPart, distance: number): number {
   402→  let dmg = w.damage
   403→  if (part === 'head') dmg *= w.headMult
   404→  else if (part === 'legs') dmg *= w.legMult
   405→  if (distance > w.falloffStart) {
   406→    const t = Math.min(1, (distance - w.falloffStart) / Math.max(1, w.falloffEnd - w.falloffStart))
   407→    dmg *= 1 - (1 - w.falloffMin) * t
   408→  }
   409→  return Math.round(dmg)
   410→}

   488→function gatedWall(ax: 'x' | 'z', at: number, from: number, to: number, H: number, T: number, mat: MatKey, gateHalf = 1.6, doorH = 0): void {
   489→  const lo = Math.min(from, to), hi = Math.max(from, to)
   490→  const mid = (lo + hi) / 2
   491→  const mk = (c: number, len: number) => {
   492→    if (len <= 0.05) return
   493→    if (ax === 'x') B(c, H / 2, at, len, H, T, mat)
   494→    else B(at, H / 2, c, T, H, len, mat)
   495→  }
   496→  mk((lo + (mid - gateHalf)) / 2, (mid - gateHalf) - lo)
   497→  mk((hi + (mid + gateHalf)) / 2, hi - (mid + gateHalf))
   498→  if (gateHalf > 0 && H > doorH + 0.6) {
   499→    const lintelH = H - Math.max(doorH, 0.01)
   500→    if (lintelH > 0.15) {
   501→      if (ax === 'x') B(mid, H - lintelH / 2, at, gateHalf * 2, lintelH, T, mat)
   502→      else B(at, H - lintelH / 2, mid, T, lintelH, gateHalf * 2, mat)
   503→    }
   504→  }
   505→}

   510→const ROT = { N: 0, E: Math.PI / 2, S: Math.PI, W: Math.PI * 1.5 } as const
   511→type Facing = keyof typeof ROT
   512→
   513→/** Caja en coordenadas locales del edificio, girada según `f` */
   514→function BR(cx: number, cz: number, f: Facing, lx: number, ly: number, lz: number, w: number, h: number, d: number, mat: MatKey): void {
   515→  const ang = ROT[f]
   516→  const c = Math.cos(ang), s = Math.sin(ang)
   517→  const rx = lx * c - lz * s
   518→  const rz = lx * s + lz * c
   519→  const swap = Math.abs(s) > 0.5
   520→  B(cx + rx, ly, cz + rz, swap ? d : w, h, swap ? w : d, mat)
   521→}

   523→interface WallOpts {
   524→  H?: number; T?: number; y0?: number
   525→  door?: number; doorHalf?: number; doorH?: number
   526→  wins?: number[]; winW?: number; bandLo?: number; bandHi?: number
   527→}
   528→
   529→/**
   530→ * Muro de edificio en coords. locales. `axis 'z'`: corre a lo largo de x en z=at;
   531→ * `axis 'x'`: corre a lo largo de z en x=at. Puerta = hueco a toda altura con dintel;
   532→ * ventanas = banda practiable (se puede disparar a través) entre zócalo y franja superior.
   533→ */
   534→function wallL(cx: number, cz: number, f: Facing, axis: 'x' | 'z', at: number, from: number, to: number, mat: MatKey, opts: WallOpts): void {
   535→  const H = opts.H ?? 3.3
   536→  const T = opts.T ?? 0.35
   537→  const y0 = opts.y0 ?? 0
   538→  const bandLo = y0 + (opts.bandLo ?? 1.45)
   539→  const bandHi = y0 + (opts.bandHi ?? 2.25)
   540→  const winW = opts.winW ?? 1.3
   541→  const lo = Math.min(from, to), hi = Math.max(from, to)
   542→  const put = (c: number, len: number, ya: number, yb: number) => {
   543→    if (len <= 0.06 || yb - ya <= 0.06) return
   544→    const cy = (ya + yb) / 2
   545→    if (axis === 'z') BR(cx, cz, f, c, cy, at, len, yb - ya, T, mat)
   546→    else BR(cx, cz, f, at, cy, c, T, yb - ya, len, mat)
   547→  }
   548→  let solids: [number, number][] = [[lo, hi]]
   549→  if (opts.door !== undefined) {
   550→    const dh = opts.doorHalf ?? 1.05
   551→    const doorH = y0 + (opts.doorH ?? 2.15)
   552→    const dLo = opts.door - dh, dHi = opts.door + dh
   553→    solids = []
   554→    if (dLo - lo > 0.05) solids.push([lo, dLo])
   555→    if (hi - dHi > 0.05) solids.push([dHi, hi])
   556→    put(opts.door, dh * 2, doorH, y0 + H)
   557→  }
   558→  for (const [a, b] of solids) {
   559→    put((a + b) / 2, b - a, y0, bandLo)
   560→    put((a + b) / 2, b - a, bandHi, y0 + H)
   561→    const gaps: [number, number][] = []
   562→    for (const w of opts.wins ?? []) {
   563→      const g0 = w - winW / 2, g1 = w + winW / 2
   564→      if (g0 > a + 0.3 && g1 < b - 0.3) gaps.push([g0, g1])
   565→    }
   566→    gaps.sort((p, q) => p[0] - q[0])
   567→    let cur = a
   568→    for (const [g0, g1] of gaps) { put((cur + g0) / 2, g0 - cur, bandLo, bandHi); cur = g1 }
   569→    put((cur + b) / 2, b - cur, bandLo, bandHi)
   570→  }
   571→}

   574→function wpTransform(cx: number, cz: number, f: Facing, lx: number, lz: number): [number, number] {
   575→  const ang = ROT[f], c = Math.cos(ang), s = Math.sin(ang)
   576→  return [cx + lx * c - lz * s, cz + lx * s + lz * c]
   577→}

   581→function stairsBR(cx: number, cz: number, f: Facing, xAt: number, zAt: number, dx: number, dz: number, steps: number, riseStep: number, runStep: number, wStair: number, y0 = 0, mat: MatKey = 'concrete'): void {
   582→  for (let i = 0; i < steps; i++) {
   583→    const rise = riseStep * (i + 1)

   587→    BR(cx, cz, f, sx, y0 + rise / 2, sz, dx !== 0 ? runStep + 0.04 : wStair, rise, dz !== 0 ? runStep + 0.04 : wStair, mat)
   588→  }

   718→// tejado accesible por escaleras exteriores N y S

   720→function mercado(cx: number, cz: number, f: Facing): void {
   721→  const MW = 10, MH = 4.6, MT = 0.6, MG = 1.7
   722→  // fachadas N y S (z = ∓MW) con puerta central
   723→  {
   724→    const segW = MW - MG
   725→    for (const sz of [-1, 1]) {
   726→      BR(cx, cz, f, -(MG + segW / 2), MH / 2, sz * MW, segW, MH, MT, 'sand')
   727→      BR(cx, cz, f, MG + segW / 2, MH / 2, sz * MW, segW, MH, MT, 'sand')
   728→      BR(cx, cz, f, 0, MH - 0.5, sz * MW, MG * 2, 1, MT, 'sand')
   729→    }
   730→    // fachadas E y O (x = ±MW) con puerta central
   731→    for (const sx of [-1, 1]) {
   732→      BR(cx, cz, f, sx * MW, MH / 2, -(MG + segW / 2), MT, MH, segW, 'sand')
   733→      BR(cx, cz, f, sx * MW, MH / 2, (MG + segW / 2), MT, MH, segW, 'sand')
   734→      BR(cx, cz, f, sx * MW, MH - 0.5, 0, MT, 1, MG * 2, 'sand')
   735→    }
   736→  }
   737→  // pilares de esquina e interiores
   738→  for (const sx of [-1, 1]) for (const sz of [-1, 1]) {
   739→    BR(cx, cz, f, sx * 9.6, MH / 2, sz * 9.6, 1, MH, 1, 'concrete')
   740→    BR(cx, cz, f, sx * 6, MH / 2, sz * 6, 0.7, MH, 0.7, 'concrete')
   741→  }
   742→  // puestos interiores
   743→  for (const sx of [-1, 1]) for (const sz of [-1, 1]) {
   744→    BR(cx, cz, f, sx * 4.2, 0.6, sz * 2, 1.2, 1.2, 1.2, 'crate')

   748→  // techo con parapeto (hueco N/S donde llegan las escaleras)
   749→  BR(cx, cz, f, 0, MH + 0.15, 0, MW * 2 + 1.4, 0.3, MW * 2 + 1.4, 'roof')

   751→  BR(cx, cz, f, -4.4, MH + 0.45, 10.3, 11.9, 0.4, 0.3, 'concrete')   // parapeto S (hueco a la der.)
   752→  BR(cx, cz, f, 10.3, MH + 0.45, 0, 0.3, 0.4, MW * 2 + 1.4, 'concrete')
   753→  BR(cx, cz, f, -10.3, MH + 0.45, 0, 0.3, 0.4, MW * 2 + 1.4, 'concrete')
   754→  // escaleras exteriores N y S (junto a la fachada, suben a lo largo de x)
   755→  stairsBR(cx, cz, f, -10.6, -12.6, 1, 0, 9, 0.5, 0.9, 1.0, 0)
   756→  stairsBR(cx, cz, f, 10.6, 12.6, -1, 0, 9, 0.5, 0.9, 1.0, 0)
   757→  WP_EXTRA.push(
   758→    wpTransform(cx, cz, f, 0, -11),
   759→    wpTransform(cx, cz, f, 0, 0),

  1248→/** ¿Choca el segmento p→q con alguna caja? (para línea de visión) */
  1249→export function segmentBlocked(px: number, py: number, pz: number, qx: number, qy: number, qz: number, boxes: AABB[] = MAP_AABBS): boolean {
  1250→  const dx = qx - px, dy = qy - py, dz = qz - pz
  1251→  for (let i = 0; i < boxes.length; i++) {
  1252→    const b = boxes[i]
  1253→    let tmin = 0, tmax = 1
  1254→    // X
  1255→    if (Math.abs(dx) < 1e-9) {
  1256→      if (px < b.minX || px > b.maxX) continue
  1257→    } else {
  1258→      let t1 = (b.minX - px) / dx, t2 = (b.maxX - px) / dx
  1259→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
  1260→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
  1261→      if (tmin > tmax) continue
  1262→    }
  1263→    // Y
  1264→    if (Math.abs(dy) < 1e-9) {
  1265→      if (py < b.minY || py > b.maxY) continue
  1266→    } else {
  1267→      let t1 = (b.minY - py) / dy, t2 = (b.maxY - py) / dy
  1268→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
  1269→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
  1270→      if (tmin > tmax) continue
  1271→    }
  1272→    // Z
  1273→    if (Math.abs(dz) < 1e-9) {
  1274→      if (pz < b.minZ || pz > b.maxZ) continue
  1275→    } else {
  1276→      let t1 = (b.minZ - pz) / dz, t2 = (b.maxZ - pz) / dz
  1277→      if (t1 > t2) { const tmp = t1; t1 = t2; t2 = tmp }
  1278→      tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2)
  1279→      if (tmin > tmax) continue
  1280→    }
  1281→    return true
  1282→  }
  1283→  return false
  1284→}

  1333→      if (!segmentBlocked(WAYPOINTS[i][0], 0.5, WAYPOINTS[i][1], WAYPOINTS[j][0], 0.5, WAYPOINTS[j][1])) {
  1334→        WAYPOINT_EDGES[i].push(j)
  1335→        WAYPOINT_EDGES[j].push(i)
  1336→      }
  1337→    }
  1338→  }
