    88→function useKeyCapture() {
    89→  const [capture, setCapture] = useState<ActionId | null>(null)
    90→  useEffect(() => {
    91→    if (!capture) return
    92→    const onKey = (e: KeyboardEvent) => {
    93→      e.preventDefault()
    94→      e.stopPropagation()
    95→      // Escape cancela · el resto reasigna
    96→      if (e.code !== 'Escape') useGame.getState().setKeybind(capture, e.code)
    97→      setCapture(null)
    98→    }
    99→    // los botones del RATÓN también se pueden asignar (disparar/apuntar)
   100→    const onMouse = (e: MouseEvent) => {
   101→      e.preventDefault()
   102→      e.stopPropagation()
   103→      useGame.getState().setKeybind(capture, `Mouse${e.button}`)
   104→      setCapture(null)
   105→    }
   106→    const onCtx = (e: Event): void => { e.preventDefault() }
   107→    window.addEventListener('keydown', onKey, true)
   108→    window.addEventListener('mousedown', onMouse, true)
   109→    window.addEventListener('contextmenu', onCtx, true)
   110→    return () => {
   111→      window.removeEventListener('keydown', onKey, true)
   112→      window.removeEventListener('mousedown', onMouse, true)
   113→      window.removeEventListener('contextmenu', onCtx, true)
   114→    }
   115→  }, [capture])
   116→  return { capture, setCapture }
   117→}

   119→/** captura de botones del MANDO (sondeo de gamepads) */
   120→function usePadCapture() {
   121→  const [padCapture, setPadCapture] = useState<PadAction | null>(null)
   122→  useEffect(() => {
   123→    if (!padCapture) return
   124→    let raf = 0
   125→    const poll = (): void => {
   126→      const pads = navigator.getGamepads?.() ?? []
   127→      for (const p of pads) {
   128→        if (!p) continue
   129→        for (let i = 0; i < p.buttons.length; i++) {
   130→          if (p.buttons[i]?.pressed) {
   131→            useGame.getState().setPadBind(padCapture, i)
   132→            setPadCapture(null)
   133→            return
   134→          }
   135→        }
   136→      }
   137→      raf = requestAnimationFrame(poll)
   138→    }
   139→    raf = requestAnimationFrame(poll)
   140→    return () => cancelAnimationFrame(raf)
   141→  }, [padCapture])
   142→  return { padCapture, setPadCapture }
   143→}
